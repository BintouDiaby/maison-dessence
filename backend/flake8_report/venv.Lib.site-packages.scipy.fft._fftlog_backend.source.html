121 def fhtoffset(dln, mu, initial=0.0, bias=0.0):
122 """Return optimal offset for a fast Hankel transform.
123  
124 Returns an offset close to `initial` that fulfils the low-ringing
125 condition of [1]_ for the fast Hankel transform `fht` with logarithmic
126 spacing `dln`, order `mu` and bias `bias`.
127  
128 Parameters
129 ----------
130 dln : float
131 Uniform logarithmic spacing of the transform.
132 mu : float
133 Order of the Hankel transform, any positive or negative real number.
134 initial : float, optional
135 Initial value for the offset. Returns the closest value that fulfils
136 the low-ringing condition.
137 bias : float, optional
138 Exponent of power law bias, any positive or negative real number.
139  
140 Returns
141 -------
142 offset : float
143 Optimal offset of the uniform logarithmic spacing of the transform that
144 fulfils a low-ringing condition.
145  
146 Examples
147 --------
148 >>> from scipy.fft import fhtoffset
149 >>> dln = 0.1
150 >>> mu = 2.0
151 >>> initial = 0.5
152 >>> bias = 0.0
153 >>> offset = fhtoffset(dln, mu, initial, bias)
154 >>> offset
155 0.5454581477676637
156  
157 See Also
158 --------
159 fht : Definition of the fast Hankel transform.
160  
161 References
162 ----------
163 .. [1] Hamilton A. J. S., 2000, MNRAS, 312, 257 (astro-ph/9905191)
164  
165 """
166  
167 lnkr, q = initial, bias
168  
169 xp = (mu+1+q)/2
170 xm = (mu+1-q)/2
171 y = np.pi/(2*dln)
172 zp = loggamma(xp + 1j*y)
173 zm = loggamma(xm + 1j*y)
174 arg = (LN_2 - lnkr)/dln + (zp.imag + zm.imag)/np.pi
175 return lnkr + (arg - np.round(arg))*dln
