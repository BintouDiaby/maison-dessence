13 def next_fast_len(target, real=False):
14 """Find the next fast size of input data to ``fft``, for zero-padding, etc.
15  
16 SciPy's FFT algorithms gain their speed by a recursive divide and conquer
17 strategy. This relies on efficient functions for small prime factors of the
18 input length. Thus, the transforms are fastest when using composites of the
19 prime factors handled by the fft implementation. If there are efficient
20 functions for all radices <= `n`, then the result will be a number `x`
21 >= ``target`` with only prime factors < `n`. (Also known as `n`-smooth
22 numbers)
23  
24 Parameters
25 ----------
26 target : int
27 Length to start searching from. Must be a positive integer.
28 real : bool, optional
29 True if the FFT involves real input or output (e.g., `rfft` or `hfft`
30 but not `fft`). Defaults to False.
31  
32 Returns
33 -------
34 out : int
35 The smallest fast length greater than or equal to ``target``.
36  
37 Notes
38 -----
39 The result of this function may change in future as performance
40 considerations change, for example, if new prime factors are added.
41  
42 Calling `fft` or `ifft` with real input data performs an ``'R2C'``
43 transform internally.
44  
45 Examples
46 --------
47 On a particular machine, an FFT of prime length takes 11.4 ms:
48  
49 >>> from scipy import fft
50 >>> import numpy as np
51 >>> rng = np.random.default_rng()
52 >>> min_len = 93059 # prime length is worst case for speed
53 >>> a = rng.standard_normal(min_len)
54 >>> b = fft.fft(a)
55  
56 Zero-padding to the next regular length reduces computation time to
57 1.6 ms, a speedup of 7.3 times:
58  
59 >>> fft.next_fast_len(min_len, real=True)
60 93312
61 >>> b = fft.fft(a, 93312)
62  
63 Rounding up to the next power of 2 is not optimal, taking 3.0 ms to
64 compute; 1.9 times longer than the size given by ``next_fast_len``:
65  
66 >>> b = fft.fft(a, 131072)
67  
68 """

257 def fftshift(x, axes=None):
258 """Shift the zero-frequency component to the center of the spectrum.
259  
260 This function swaps half-spaces for all axes listed (defaults to all).
261 Note that ``y[0]`` is the Nyquist component only if ``len(x)`` is even.
262  
263 Parameters
264 ----------
265 x : array_like
266 Input array.
267 axes : int or shape tuple, optional
268 Axes over which to shift. Default is None, which shifts all axes.
269  
270 Returns
271 -------
272 y : ndarray
273 The shifted array.
274  
275 See Also
276 --------
277 ifftshift : The inverse of `fftshift`.
278  
279 Examples
280 --------
281 >>> import numpy as np
282 >>> freqs = np.fft.fftfreq(10, 0.1)
283 >>> freqs
284 array([ 0., 1., 2., ..., -3., -2., -1.])
285 >>> np.fft.fftshift(freqs)
286 array([-5., -4., -3., -2., -1., 0., 1., 2., 3., 4.])
287  
288 Shift the zero-frequency component only along the second axis:
289  
290 >>> freqs = np.fft.fftfreq(9, d=1./9).reshape(3, 3)
291 >>> freqs
292 array([[ 0., 1., 2.],
293 [ 3., 4., -4.],
294 [-3., -2., -1.]])
295 >>> np.fft.fftshift(freqs, axes=(1,))
296 array([[ 2., 0., 1.],
297 [-4., 3., 4.],
298 [-1., -3., -2.]])
299  
300 """
301 xp = array_namespace(x)
302 if hasattr(xp, 'fft'):
303 return xp.fft.fftshift(x, axes=axes)
304 x = np.asarray(x)
305 y = np.fft.fftshift(x, axes=axes)
306 return xp.asarray(y)
