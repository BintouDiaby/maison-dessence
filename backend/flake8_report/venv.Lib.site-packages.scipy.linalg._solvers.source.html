31 @_apply_over_batch(('a', 2), ('b', 2), ('q', 2))
32 def solve_sylvester(a, b, q):
33 """
34 Computes a solution (X) to the Sylvester equation :math:`AX + XB = Q`.
35  
36 Parameters
37 ----------
38 a : (M, M) array_like
39 Leading matrix of the Sylvester equation
40 b : (N, N) array_like
41 Trailing matrix of the Sylvester equation
42 q : (M, N) array_like
43 Right-hand side
44  
45 Returns
46 -------
47 x : (M, N) ndarray
48 The solution to the Sylvester equation.
49  
50 Raises
51 ------
52 LinAlgError
53 If solution was not found
54  
55 Notes
56 -----
57 Computes a solution to the Sylvester matrix equation via the Bartels-
58 Stewart algorithm. The A and B matrices first undergo Schur
59 decompositions. The resulting matrices are used to construct an
60 alternative Sylvester equation (``RY + YS^T = F``) where the R and S
61 matrices are in quasi-triangular form (or, when R, S or F are complex,
62 triangular form). The simplified equation is then solved using
63 ``*TRSYL`` from LAPACK directly.
64  
65 .. versionadded:: 0.11.0
66  
67 Examples
68 --------
69 Given `a`, `b`, and `q` solve for `x`:
70  
71 >>> import numpy as np
72 >>> from scipy import linalg
73 >>> a = np.array([[-3, -2, 0], [-1, -1, 3], [3, -5, -1]])
74 >>> b = np.array([[1]])
75 >>> q = np.array([[1],[2],[3]])
76 >>> x = linalg.solve_sylvester(a, b, q)
77 >>> x
78 array([[ 0.0625],
79 [-0.5625],
80 [ 0.6875]])
81 >>> np.allclose(a.dot(x) + x.dot(b), q)
82 True
83  
84 """
85 # Accommodate empty a
86 if a.size == 0 or b.size == 0:
87 tdict = {'s': np.float32, 'd': np.float64,
88 'c': np.complex64, 'z': np.complex128}
89 func, = get_lapack_funcs(('trsyl',), arrays=(a, b, q))
90 return np.empty(q.shape, dtype=tdict[func.typecode])
91  
92 # Compute the Schur decomposition form of a
93 r, u = schur(a, output='real')
94  
95 # Compute the Schur decomposition of b
96 s, v = schur(b.conj().transpose(), output='real')
97  
98 # Construct f = u'*q*v
99 f = np.dot(np.dot(u.conj().transpose(), q), v)
100  
101 # Call the Sylvester equation solver
102 trsyl, = get_lapack_funcs(('trsyl',), (r, s, f))
103 if trsyl is None:
104 raise RuntimeError('LAPACK implementation does not contain a proper '
105 'Sylvester equation solver (TRSYL)')
106 y, scale, info = trsyl(r, s, f, tranb='C')
107  
108 y = scale*y
109  
110 if info < 0:
111 raise LinAlgError(f"Illegal value encountered in the {-info} term")
112  
113 return np.dot(np.dot(u, y), v.conj().transpose())

116 @_apply_over_batch(('a', 2), ('q', 2))
117 def solve_continuous_lyapunov(a, q):
118 """
119 Solves the continuous Lyapunov equation :math:`AX + XA^H = Q`.
120  
121 Uses the Bartels-Stewart algorithm to find :math:`X`.
122  
123 Parameters
124 ----------
125 a : array_like
126 A square matrix
127  
128 q : array_like
129 Right-hand side square matrix
130  
131 Returns
132 -------
133 x : ndarray
134 Solution to the continuous Lyapunov equation
135  
136 See Also
137 --------
138 solve_discrete_lyapunov : computes the solution to the discrete-time
139 Lyapunov equation
140 solve_sylvester : computes the solution to the Sylvester equation
141  
142 Notes
143 -----
144 The continuous Lyapunov equation is a special form of the Sylvester
145 equation, hence this solver relies on LAPACK routine ?TRSYL.
146  
147 .. versionadded:: 0.11.0
148  
149 Examples
150 --------
151 Given `a` and `q` solve for `x`:
152  
153 >>> import numpy as np
154 >>> from scipy import linalg
155 >>> a = np.array([[-3, -2, 0], [-1, -1, 0], [0, -5, -1]])
156 >>> b = np.array([2, 4, -1])
157 >>> q = np.eye(3)
158 >>> x = linalg.solve_continuous_lyapunov(a, q)
159 >>> x
160 array([[ -0.75 , 0.875 , -3.75 ],
161 [ 0.875 , -1.375 , 5.3125],
162 [ -3.75 , 5.3125, -27.0625]])
163 >>> np.allclose(a.dot(x) + x.dot(a.T), q)
164 True
165 """
166  
167 a = np.atleast_2d(_asarray_validated(a, check_finite=True))
168 q = np.atleast_2d(_asarray_validated(q, check_finite=True))
169  
170 r_or_c = float
171  
172 for ind, _ in enumerate((a, q)):
173 if np.iscomplexobj(_):
174 r_or_c = complex
175  
176 if not np.equal(*_.shape):
177 raise ValueError(f"Matrix {'aq'[ind]} should be square.")
178  
179 # Shape consistency check
180 if a.shape != q.shape:
181 raise ValueError("Matrix a and q should have the same shape.")
182  
183 # Accommodate empty array
184 if a.size == 0:
185 tdict = {'s': np.float32, 'd': np.float64,
186 'c': np.complex64, 'z': np.complex128}
187 func, = get_lapack_funcs(('trsyl',), arrays=(a, q))
188 return np.empty(a.shape, dtype=tdict[func.typecode])
189  
190 # Compute the Schur decomposition form of a
191 r, u = schur(a, output='real')
192  
193 # Construct f = u'*q*u
194 f = u.conj().T.dot(q.dot(u))
195  
196 # Call the Sylvester equation solver
197 trsyl = get_lapack_funcs('trsyl', (r, f))
198  
199 dtype_string = 'T' if r_or_c is float else 'C'
200 y, scale, info = trsyl(r, r, f, tranb=dtype_string)
201  
202 if info < 0:
203 raise ValueError('?TRSYL exited with the internal error '
204 f'"illegal value in argument number {-info}.". See '
205 'LAPACK documentation for the ?TRSYL error codes.')
206 elif info == 1:
207 warnings.warn('Input "a" has an eigenvalue pair whose sum is '
208 'very close to or exactly zero. The solution is '
209 'obtained via perturbing the coefficients.',
210 RuntimeWarning, stacklevel=2)
211 y *= scale
212  
213 return u.dot(y).dot(u.conj().T)

220 def _solve_discrete_lyapunov_direct(a, q):
221 """
222 Solves the discrete Lyapunov equation directly.
223  
224 This function is called by the `solve_discrete_lyapunov` function with
225 `method=direct`. It is not supposed to be called directly.
226 """
227  
228 lhs = np.kron(a, a.conj())
229 lhs = np.eye(lhs.shape[0]) - lhs
230 x = solve(lhs, q.flatten())
231  
232 return np.reshape(x, q.shape)

235 def _solve_discrete_lyapunov_bilinear(a, q):
236 """
237 Solves the discrete Lyapunov equation using a bilinear transformation.
238  
239 This function is called by the `solve_discrete_lyapunov` function with
240 `method=bilinear`. It is not supposed to be called directly.
241 """
242 eye = np.eye(a.shape[0])
243 aH = a.conj().transpose()
244 aHI_inv = inv(aH + eye)
245 b = np.dot(aH - eye, aHI_inv)
246 c = 2*np.dot(np.dot(inv(a + eye), q), aHI_inv)
247 return solve_lyapunov(b.conj().transpose(), -c)

250 @_apply_over_batch(('a', 2), ('q', 2))
251 def solve_discrete_lyapunov(a, q, method=None):
252 """
253 Solves the discrete Lyapunov equation :math:`AXA^H - X + Q = 0`.
254  
255 Parameters
256 ----------
257 a, q : (M, M) array_like
258 Square matrices corresponding to A and Q in the equation
259 above respectively. Must have the same shape.
260  
261 method : {'direct', 'bilinear'}, optional
262 Type of solver.
263  
264 If not given, chosen to be ``direct`` if ``M`` is less than 10 and
265 ``bilinear`` otherwise.
266  
267 Returns
268 -------
269 x : ndarray
270 Solution to the discrete Lyapunov equation
271  
272 See Also
273 --------
274 solve_continuous_lyapunov : computes the solution to the continuous-time
275 Lyapunov equation
276  
277 Notes
278 -----
279 This section describes the available solvers that can be selected by the
280 'method' parameter. The default method is *direct* if ``M`` is less than 10
281 and ``bilinear`` otherwise.
282  
283 Method *direct* uses a direct analytical solution to the discrete Lyapunov
  • E501 Line too long (80 > 79 characters)
284 equation. The algorithm is given in, for example, [1]_. However, it requires
285 the linear solution of a system with dimension :math:`M^2` so that
286 performance degrades rapidly for even moderately sized matrices.
287  
288 Method *bilinear* uses a bilinear transformation to convert the discrete
289 Lyapunov equation to a continuous Lyapunov equation :math:`(BX+XB'=-C)`
290 where :math:`B=(A-I)(A+I)^{-1}` and
291 :math:`C=2(A' + I)^{-1} Q (A + I)^{-1}`. The continuous equation can be
292 efficiently solved since it is a special case of a Sylvester equation.
293 The transformation algorithm is from Popov (1964) as described in [2]_.
294  
295 .. versionadded:: 0.11.0
296  
297 References
298 ----------
299 .. [1] "Lyapunov equation", Wikipedia,
300 https://en.wikipedia.org/wiki/Lyapunov_equation#Discrete_time
301 .. [2] Gajic, Z., and M.T.J. Qureshi. 2008.
302 Lyapunov Matrix Equation in System Stability and Control.
303 Dover Books on Engineering Series. Dover Publications.
304  
305 Examples
306 --------
307 Given `a` and `q` solve for `x`:
308  
309 >>> import numpy as np
310 >>> from scipy import linalg
311 >>> a = np.array([[0.2, 0.5],[0.7, -0.9]])
312 >>> q = np.eye(2)
313 >>> x = linalg.solve_discrete_lyapunov(a, q)
314 >>> x
315 array([[ 0.70872893, 1.43518822],
316 [ 1.43518822, -2.4266315 ]])
317 >>> np.allclose(a.dot(x).dot(a.T)-x, -q)
318 True
319  
320 """
321 a = np.asarray(a)
322 q = np.asarray(q)
323 if method is None:
324 # Select automatically based on size of matrices
325 if a.shape[0] >= 10:
326 method = 'bilinear'
327 else:
328 method = 'direct'
329  
330 meth = method.lower()
331  
332 if meth == 'direct':
333 x = _solve_discrete_lyapunov_direct(a, q)
334 elif meth == 'bilinear':
335 x = _solve_discrete_lyapunov_bilinear(a, q)
336 else:
337 raise ValueError(f'Unknown solver {method}')
338  
339 return x

342 def solve_continuous_are(a, b, q, r, e=None, s=None, balanced=True):
343 r"""
344 Solves the continuous-time algebraic Riccati equation (CARE).
345  
346 The CARE is defined as
347  
348 .. math::
349  
350 X A + A^H X - X B R^{-1} B^H X + Q = 0
351  
352 The limitations for a solution to exist are :
353  
354 * All eigenvalues of :math:`A` on the right half plane, should be
355 controllable.
356  
357 * The associated hamiltonian pencil (See Notes), should have
358 eigenvalues sufficiently away from the imaginary axis.
359  
360 Moreover, if ``e`` or ``s`` is not precisely ``None``, then the
361 generalized version of CARE
362  
363 .. math::
364  
365 E^HXA + A^HXE - (E^HXB + S) R^{-1} (B^HXE + S^H) + Q = 0
366  
367 is solved. When omitted, ``e`` is assumed to be the identity and ``s``
368 is assumed to be the zero matrix with sizes compatible with ``a`` and
369 ``b``, respectively.
370  
371 The documentation is written assuming array arguments are of specified
  • E501 Line too long (82 > 79 characters)
372 "core" shapes. However, array argument(s) of this function may have additional
  • E501 Line too long (86 > 79 characters)
373 "batch" dimensions prepended to the core shape. In this case, the array is treated
  • E501 Line too long (80 > 79 characters)
374 as a batch of lower-dimensional slices; see :ref:`linalg_batch` for details.
375  
376 Parameters
377 ----------
378 a : (M, M) array_like
379 Square matrix
380 b : (M, N) array_like
381 Input
382 q : (M, M) array_like
383 Input
384 r : (N, N) array_like
385 Nonsingular square matrix
386 e : (M, M) array_like, optional
387 Nonsingular square matrix
388 s : (M, N) array_like, optional
389 Input
390 balanced : bool, optional
391 The boolean that indicates whether a balancing step is performed
392 on the data. The default is set to True.
393  
394 Returns
395 -------
396 x : (M, M) ndarray
397 Solution to the continuous-time algebraic Riccati equation.
398  
399 Raises
400 ------
401 LinAlgError
402 For cases where the stable subspace of the pencil could not be
403 isolated. See Notes section and the references for details.
404  
405 See Also
406 --------
407 solve_discrete_are : Solves the discrete-time algebraic Riccati equation
408  
409 Notes
410 -----
411 The equation is solved by forming the extended hamiltonian matrix pencil,
412 as described in [1]_, :math:`H - \lambda J` given by the block matrices ::
413  
414 [ A 0 B ] [ E 0 0 ]
415 [-Q -A^H -S ] - \lambda * [ 0 E^H 0 ]
416 [ S^H B^H R ] [ 0 0 0 ]
417  
418 and using a QZ decomposition method.
419  
420 In this algorithm, the fail conditions are linked to the symmetry
421 of the product :math:`U_2 U_1^{-1}` and condition number of
422 :math:`U_1`. Here, :math:`U` is the 2m-by-m matrix that holds the
423 eigenvectors spanning the stable subspace with 2-m rows and partitioned
424 into two m-row matrices. See [1]_ and [2]_ for more details.
425  
426 In order to improve the QZ decomposition accuracy, the pencil goes
427 through a balancing step where the sum of absolute values of
428 :math:`H` and :math:`J` entries (after removing the diagonal entries of
429 the sum) is balanced following the recipe given in [3]_.
430  
431 .. versionadded:: 0.11.0
432  
433 References
434 ----------
435 .. [1] P. van Dooren , "A Generalized Eigenvalue Approach For Solving
436 Riccati Equations.", SIAM Journal on Scientific and Statistical
437 Computing, Vol.2(2), :doi:`10.1137/0902010`
438  
439 .. [2] A.J. Laub, "A Schur Method for Solving Algebraic Riccati
440 Equations.", Massachusetts Institute of Technology. Laboratory for
441 Information and Decision Systems. LIDS-R ; 859. Available online :
442 http://hdl.handle.net/1721.1/1301
443  
444 .. [3] P. Benner, "Symplectic Balancing of Hamiltonian Matrices", 2001,
445 SIAM J. Sci. Comput., 2001, Vol.22(5), :doi:`10.1137/S1064827500367993`
446  
447 Examples

452 >>> from scipy import linalg

460 [ 14.48528137, 9.65685425]])

462 True
463  
464 """

469 @_apply_over_batch(('a', 2), ('b', 2), ('q', 2), ('r', 2), ('e', 2), ('s', 2))
470 def _solve_continuous_are(a, b, q, r, e, s, balanced):
471 # Validate input arguments
472 a, b, q, r, e, s, m, n, r_or_c, gen_are = _are_validate_args(
473 a, b, q, r, e, s, 'care')
474  
475 H = np.empty((2*m+n, 2*m+n), dtype=r_or_c)
476 H[:m, :m] = a
477 H[:m, m:2*m] = 0.
478 H[:m, 2*m:] = b
479 H[m:2*m, :m] = -q
480 H[m:2*m, m:2*m] = -a.conj().T
481 H[m:2*m, 2*m:] = 0. if s is None else -s
482 H[2*m:, :m] = 0. if s is None else s.conj().T
483 H[2*m:, m:2*m] = b.conj().T
484 H[2*m:, 2*m:] = r
485  
486 if gen_are and e is not None:
487 J = block_diag(e, e.conj().T, np.zeros_like(r, dtype=r_or_c))
488 else:
489 J = block_diag(np.eye(2*m), np.zeros_like(r, dtype=r_or_c))
490  
491 if balanced:
492 # xGEBAL does not remove the diagonals before scaling. Also
493 # to avoid destroying the Symplectic structure, we follow Ref.3
494 M = np.abs(H) + np.abs(J)
495 np.fill_diagonal(M, 0.)
496 _, (sca, _) = matrix_balance(M, separate=1, permute=0)
497 # do we need to bother?
498 if not np.allclose(sca, np.ones_like(sca)):
499 # Now impose diag(D,inv(D)) from Benner where D is
500 # square root of s_i/s_(n+i) for i=0,....
501 sca = np.log2(sca)
502 # NOTE: Py3 uses "Bankers Rounding: round to the nearest even" !!
503 s = np.round((sca[m:2*m] - sca[:m])/2)
504 sca = 2 ** np.r_[s, -s, sca[2*m:]]
505 # Elementwise multiplication via broadcasting.
506 elwisescale = sca[:, None] * np.reciprocal(sca)
507 H *= elwisescale
508 J *= elwisescale
509  
510 # Deflate the pencil to 2m x 2m ala Ref.1, eq.(55)
511 q, r = qr(H[:, -n:])
512 H = q[:, n:].conj().T.dot(H[:, :2*m])
513 J = q[:2*m, n:].conj().T.dot(J[:2*m, :2*m])
514  
515 # Decide on which output type is needed for QZ
516 out_str = 'real' if r_or_c is float else 'complex'
517  
518 _, _, _, _, _, u = ordqz(H, J, sort='lhp', overwrite_a=True,
519 overwrite_b=True, check_finite=False,
520 output=out_str)
521  
522 # Get the relevant parts of the stable subspace basis
523 if e is not None:
524 u, _ = qr(np.vstack((e.dot(u[:m, :m]), u[m:, :m])))
525 u00 = u[:m, :m]
526 u10 = u[m:, :m]
527  
528 # Solve via back-substituion after checking the condition of u00
529 up, ul, uu = lu(u00)
530 if 1/cond(uu) < np.spacing(1.):
531 raise LinAlgError('Failed to find a finite solution.')
532  
533 # Exploit the triangular structure
534 x = solve_triangular(ul.conj().T,
535 solve_triangular(uu.conj().T,
536 u10.conj().T,
537 lower=True),
538 unit_diagonal=True,
539 ).conj().T.dot(up.conj().T)
540 if balanced:
541 x *= sca[:m, None] * sca[:m]
542  
543 # Check the deviation from symmetry for lack of success
544 # See proof of Thm.5 item 3 in [2]
545 u_sym = u00.conj().T.dot(u10)
546 n_u_sym = norm(u_sym, 1)
547 u_sym = u_sym - u_sym.conj().T
548 sym_threshold = np.max([np.spacing(1000.), 0.1*n_u_sym])
549  
550 if norm(u_sym, 1) > sym_threshold:
551 raise LinAlgError('The associated Hamiltonian pencil has eigenvalues '
552 'too close to the imaginary axis')
553  
554 return (x + x.conj().T)/2

557 def solve_discrete_are(a, b, q, r, e=None, s=None, balanced=True):
558 r"""
559 Solves the discrete-time algebraic Riccati equation (DARE).
