1 """
2 Low-level BLAS functions (:mod:`scipy.linalg.blas`)
3 ===================================================
4  
5 This module contains low-level functions from the BLAS library.
6  
7 .. versionadded:: 0.12.0
8  
9 .. note::
10  
11 The common ``overwrite_<>`` option in many routines, allows the
12 input arrays to be overwritten to avoid extra memory allocation.
13 However this requires the array to satisfy two conditions
14 which are memory order and the data type to match exactly the
15 order and the type expected by the routine.
16  
17 As an example, if you pass a double precision float array to any
18 ``S....`` routine which expects single precision arguments, f2py
19 will create an intermediate array to match the argument types and
20 overwriting will be performed on that intermediate array.
21  
22 Similarly, if a C-contiguous array is passed, f2py will pass a
23 FORTRAN-contiguous array internally. Please make sure that these
24 details are satisfied. More information can be found in the f2py
25 documentation.
26  
27 .. warning::
28  
29 These functions do little to no error checking.
30 It is possible to cause crashes by mis-using them,
31 so prefer using the higher-level routines in `scipy.linalg`.

270 def find_best_blas_type(arrays=(), dtype=None):
271 """Find best-matching BLAS/LAPACK type.
272  
273 Arrays are used to determine the optimal prefix of BLAS routines.
274  
275 Parameters
276 ----------
277 arrays : sequence of ndarrays, optional
278 Arrays can be given to determine optimal prefix of BLAS
279 routines. If not given, double-precision routines will be
280 used, otherwise the most generic type in arrays will be used.
281 dtype : str or dtype, optional
282 Data-type specifier. Not used if `arrays` is non-empty.
283  
284 Returns
285 -------
286 prefix : str
287 BLAS/LAPACK prefix character.
288 dtype : dtype
289 Inferred Numpy data type.
290 prefer_fortran : bool
291 Whether to prefer Fortran order routines over C order.
292  
293 Examples
294 --------
295 >>> import numpy as np
296 >>> import scipy.linalg.blas as bla
297 >>> rng = np.random.default_rng()
298 >>> a = rng.random((10,15))
299 >>> b = np.asfortranarray(a) # Change the memory layout order
300 >>> bla.find_best_blas_type((a,))
301 ('d', dtype('float64'), False)
302 >>> bla.find_best_blas_type((a*1j,))
303 ('z', dtype('complex128'), False)
304 >>> bla.find_best_blas_type((b,))
305 ('d', dtype('float64'), True)
306  
307 """
308 dtype = np.dtype(dtype)
309 max_score = _type_score.get(dtype.char, 5)
310 prefer_fortran = False
311  
312 if arrays:
313 # In most cases, single element is passed through, quicker route
314 if len(arrays) == 1:
315 max_score = _type_score.get(arrays[0].dtype.char, 5)
316 prefer_fortran = arrays[0].flags['FORTRAN']
317 else:
318 # use the most generic type in arrays
319 scores = [_type_score.get(x.dtype.char, 5) for x in arrays]
320 max_score = max(scores)
321 ind_max_score = scores.index(max_score)
322 # safe upcasting for mix of float64 and complex64 --> prefix 'z'
323 if max_score == 3 and (2 in scores):
324 max_score = 4
325  
326 if arrays[ind_max_score].flags['FORTRAN']:
327 # prefer Fortran for leading array with column major order
328 prefer_fortran = True
329  
330 # Get the LAPACK prefix and the corresponding dtype if not fall back
331 # to 'd' and double precision float.
332 prefix, dtype = _type_conv.get(max_score, ('d', np.dtype('float64')))
333  
334 return prefix, dtype, prefer_fortran

337 def _get_funcs(names, arrays, dtype,
338 lib_name, fmodule, cmodule,
339 fmodule_name, cmodule_name, alias,
340 ilp64=False):
341 """
342 Return available BLAS/LAPACK functions.
343  
344 Used also in lapack.py. See get_blas_funcs for docstring.
345 """
346  
347 funcs = []
348 unpack = False
349 dtype = np.dtype(dtype)
350 module1 = (cmodule, cmodule_name)
351 module2 = (fmodule, fmodule_name)
352  
353 if isinstance(names, str):
354 names = (names,)
355 unpack = True
356  
357 prefix, dtype, prefer_fortran = find_best_blas_type(arrays, dtype)
358  
359 if prefer_fortran:
360 module1, module2 = module2, module1
361  
362 for name in names:
363 func_name = prefix + name
364 func_name = alias.get(func_name, func_name)
365 func = getattr(module1[0], func_name, None)
366 module_name = module1[1]
367 if func is None:
368 func = getattr(module2[0], func_name, None)
369 module_name = module2[1]
370 if func is None:
371 raise ValueError(
372 f'{lib_name} function {func_name} could not be found')
373 func.module_name, func.typecode = module_name, prefix
374 func.dtype = dtype
375 if not ilp64:
376 func.int_dtype = np.dtype(np.intc)
377 else:
378 func.int_dtype = np.dtype(np.int64)
379 func.prefix = prefix # Backward compatibility
380 funcs.append(func)
381  
382 if unpack:
383 return funcs[0]
384 else:
385 return funcs

389 """
390 Memoized fast path for _get_funcs instances

395 @functools.wraps(func)

397 key = (names, dtype, ilp64)
398 for array in arrays:
399 # cf. find_blas_funcs
400 key += (array.dtype.char, array.flags.fortran)

402 try:
403 value = memo.get(key)
404 except TypeError:
405 # unhashable key etc.
406 key = None
407 value = None
408  
409 if value is not None:
410 return value
411  
412 value = func(names, arrays, dtype, ilp64)
413  
414 if key is not None:
415 memo[key] = value
416  
417 return value

423 def get_blas_funcs(names, arrays=(), dtype=None, ilp64=False):
424 """Return available BLAS function objects from names.
425  
426 Arrays are used to determine the optimal prefix of BLAS routines.
427  
428 Parameters
429 ----------
430 names : str or sequence of str
431 Name(s) of BLAS functions without type prefix.
432  
433 arrays : sequence of ndarrays, optional
434 Arrays can be given to determine optimal prefix of BLAS
435 routines. If not given, double-precision routines will be
436 used, otherwise the most generic type in arrays will be used.
437  
438 dtype : str or dtype, optional
439 Data-type specifier. Not used if `arrays` is non-empty.
440  
441 ilp64 : {True, False, 'preferred'}, optional
442 Whether to return ILP64 routine variant.
443 Choosing 'preferred' returns ILP64 routine if available,
444 and otherwise the 32-bit routine. Default: False
445  
446 Returns
447 -------
448 funcs : list
449 List containing the found function(s).
450  
451  
452 Notes
453 -----
454 This routine automatically chooses between Fortran/C
455 interfaces. Fortran code is used whenever possible for arrays with
456 column major order. In all other cases, C code is preferred.
457  
458 In BLAS, the naming convention is that all functions start with a
459 type prefix, which depends on the type of the principal
460 matrix. These can be one of {'s', 'd', 'c', 'z'} for the NumPy
461 types {float32, float64, complex64, complex128} respectively.
462 The code and the dtype are stored in attributes `typecode` and `dtype`
463 of the returned functions.
464  
465 Examples
466 --------
467 >>> import numpy as np
468 >>> import scipy.linalg as LA
469 >>> rng = np.random.default_rng()
470 >>> a = rng.random((3,2))
471 >>> x_gemv = LA.get_blas_funcs('gemv', (a,))
472 >>> x_gemv.typecode
473 'd'
474 >>> x_gemv = LA.get_blas_funcs('gemv',(a*1j,))
475 >>> x_gemv.typecode
476 'z'
477  
478 """
479 if isinstance(ilp64, str):
480 if ilp64 == 'preferred':
481 ilp64 = HAS_ILP64
482 else:
483 raise ValueError("Invalid value for 'ilp64'")
484  
485 if not ilp64:
486 return _get_funcs(names, arrays, dtype,
487 "BLAS", _fblas, _cblas, "fblas", "cblas",
488 _blas_alias, ilp64=False)
489 else:
490 if not HAS_ILP64:
491 raise RuntimeError("BLAS ILP64 routine requested, but Scipy "
492 "compiled only with 32-bit BLAS")
493 return _get_funcs(names, arrays, dtype,
494 "BLAS", _fblas_64, None, "fblas_64", None,
495 _blas_alias, ilp64=True)
