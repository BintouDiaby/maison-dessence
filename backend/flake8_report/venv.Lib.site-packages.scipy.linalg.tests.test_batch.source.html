19 def get_nearly_hermitian(shape, dtype, atol, rng):
20 # Generate a batch of nearly Hermitian matrices with specified
21 # `shape` and `dtype`. `atol` controls the level of noise in
22 # Hermitian-ness to by generated by `rng`.
23 A = rng.random(shape).astype(dtype)
24 At = np.conj(A.swapaxes(-1, -2))
25 noise = rng.standard_normal(size=A.shape).astype(dtype) * atol
