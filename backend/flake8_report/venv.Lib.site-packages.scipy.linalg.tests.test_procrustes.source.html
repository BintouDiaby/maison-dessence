71 def test_orthogonal_procrustes():
72 rng = np.random.RandomState(1234)
73 for m, n in ((6, 4), (4, 4), (4, 6)):
74 # Sample a random target matrix.
75 B = rng.randn(m, n)
76 # Sample a random orthogonal matrix
77 # by computing eigh of a sampled symmetric matrix.
78 X = rng.randn(n, n)
79 w, V = eigh(X.T + X)
80 assert_allclose(inv(V), V.T)
81 # Compute a matrix with a known orthogonal transformation that gives B.
82 A = np.dot(B, V.T)
83 # Check that an orthogonal transformation from A to B can be recovered.
84 R, s = orthogonal_procrustes(A, B)
85 assert_allclose(inv(R), R.T)
86 assert_allclose(A.dot(R), B)
87 # Create a perturbed input matrix.
88 A_perturbed = A + 1e-2 * rng.randn(m, n)
89 # Check that the orthogonal procrustes function can find an orthogonal
90 # transformation that is better than the orthogonal transformation
91 # computed from the original input matrix.
92 R_prime, s = orthogonal_procrustes(A_perturbed, B)
93 assert_allclose(inv(R_prime), R_prime.T)
94 # Compute the naive and optimal transformations of the perturbed input.
95 naive_approx = A_perturbed.dot(R)
96 optim_approx = A_perturbed.dot(R_prime)
97 # Compute the Frobenius norm errors of the matrix approximations.
98 naive_approx_error = norm(naive_approx - B, ord='fro')
99 optim_approx_error = norm(optim_approx - B, ord='fro')
100 # Check that the orthogonal Procrustes approximation is better.
101 assert_array_less(optim_approx_error, naive_approx_error)

109 def test_orthogonal_procrustes_exact_example():
110 # Check a small application.
111 # It uses translation, scaling, reflection, and rotation.
112 #
113 # |
114 # a b |
115 # |
116 # d c | w
117 # |
118 # --------+--- x ----- z ---
119 # |
120 # | y
121 # |
122 #
123 A_orig = np.array([[-3, 3], [-2, 3], [-2, 2], [-3, 2]], dtype=float)
124 B_orig = np.array([[3, 2], [1, 0], [3, -2], [5, 0]], dtype=float)
125 A, A_mu = _centered(A_orig)
126 B, B_mu = _centered(B_orig)
127 R, s = orthogonal_procrustes(A, B)
128 scale = s / np.square(norm(A))
129 B_approx = scale * np.dot(A, R) + B_mu
130 assert_allclose(B_approx, B_orig, atol=1e-8)

198 @pytest.mark.parametrize('shape', [(4, 5), (5, 5), (5, 4)])
199 def test_unitary(shape):
200 # gh-12071 added support for unitary matrices; check that it
201 # works as intended.
202 m, n = shape
203 rng = np.random.default_rng(589234981235)
204 A = rng.random(shape) + rng.random(shape) * 1j
205 Q = rng.random((n, n)) + rng.random((n, n)) * 1j
206 Q, _ = np.linalg.qr(Q)
207 B = A @ Q
208 R, scale = orthogonal_procrustes(A, B)
209 assert_allclose(R @ R.conj().T, np.eye(n), atol=1e-14)
210 assert_allclose(A @ Q, B)
211 if shape != (4, 5): # solution is unique
212 assert_allclose(R, Q)
213 _, s, _ = svd(A.conj().T @ B)
214 assert_allclose(scale, np.sum(s))
