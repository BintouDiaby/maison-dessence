23

27  
28 Here the vector of independent variables x is passed as ndarray of shape

196 def residual(self, x):
197 """
198 Calculate the residual between the constraint function and the limits
199  
200 For a linear constraint of the form::
201  
202 lb <= A@x <= ub
203  
204 the lower and upper residuals between ``A@x`` and the limits are values
205 ``sl`` and ``sb`` such that::
206  
207 lb + sl == A@x == ub - sb
208  
209 When all elements of ``sl`` and ``sb`` are positive, all elements of
210 the constraint are satisfied; a negative element in ``sl`` or ``sb``
211 indicates that the corresponding element of the constraint is not
212 satisfied.
213  
214 Parameters
215 ----------
216 x: array_like
217 Vector of independent variables
218  
219 Returns
220 -------
221 sl, sb : array-like
222 The lower and upper residuals
223 """
224 return self.A@x - self.lb, self.ub - self.A@x

384
