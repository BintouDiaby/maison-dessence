13 def _constraints_to_components(constraints):
14 """
15 Convert sequence of constraints to a single set of components A, b_l, b_u.
16  
17 `constraints` could be
18  
19 1. A LinearConstraint
20 2. A tuple representing a LinearConstraint
21 3. An invalid object
22 4. A sequence of composed entirely of objects of type 1/2
23 5. A sequence containing at least one object of type 3
24  
25 We want to accept 1, 2, and 4 and reject 3 and 5.
26 """
27 message = ("`constraints` (or each element within `constraints`) must be "
28 "convertible into an instance of "
29 "`scipy.optimize.LinearConstraint`.")
30 As = []
31 b_ls = []
32 b_us = []
33  
34 # Accept case 1 by standardizing as case 4
35 if isinstance(constraints, LinearConstraint):
36 constraints = [constraints]
37 else:
38 # Reject case 3
39 try:
40 iter(constraints)
41 except TypeError as exc:
42 raise ValueError(message) from exc
43  
44 # Accept case 2 by standardizing as case 4
45 if len(constraints) == 3:
46 # argument could be a single tuple representing a LinearConstraint
47 try:
48 constraints = [LinearConstraint(*constraints)]
49 except (TypeError, ValueError, VisibleDeprecationWarning):
50 # argument was not a tuple representing a LinearConstraint
51 pass
52  
53 # Address cases 4/5
54 for constraint in constraints:
55 # if it's not a LinearConstraint or something that represents a
56 # LinearConstraint at this point, it's invalid
57 if not isinstance(constraint, LinearConstraint):
58 try:
59 constraint = LinearConstraint(*constraint)
60 except TypeError as exc:
61 raise ValueError(message) from exc
62 As.append(csc_array(constraint.A))
63 b_ls.append(np.atleast_1d(constraint.lb).astype(np.float64))
64 b_us.append(np.atleast_1d(constraint.ub).astype(np.float64))
65  
66 if len(As) > 1:
67 A = vstack(As, format="csc")
68 b_l = np.concatenate(b_ls)
69 b_u = np.concatenate(b_us)
70 else: # avoid unnecessary copying
71 A = As[0]
72 b_l = b_ls[0]
73 b_u = b_us[0]
74  
75 return A, b_l, b_u

78 def _milp_iv(c, integrality, bounds, constraints, options):
79 # objective IV
80 if issparse(c):
81 raise ValueError("`c` must be a dense array.")
82 c = np.atleast_1d(c).astype(np.float64)
83 if c.ndim != 1 or c.size == 0 or not np.all(np.isfinite(c)):
84 message = ("`c` must be a one-dimensional array of finite numbers "
85 "with at least one element.")
86 raise ValueError(message)
87  
88 # integrality IV
89 if issparse(integrality):
90 raise ValueError("`integrality` must be a dense array.")
91 message = ("`integrality` must contain integers 0-3 and be broadcastable "
92 "to `c.shape`.")
93 if integrality is None:
94 integrality = 0
95 try:
96 integrality = np.broadcast_to(integrality, c.shape).astype(np.uint8)
97 except ValueError:
98 raise ValueError(message)
99 if integrality.min() < 0 or integrality.max() > 3:
100 raise ValueError(message)
101  
102 # bounds IV
103 if bounds is None:
104 bounds = Bounds(0, np.inf)
105 elif not isinstance(bounds, Bounds):
106 message = ("`bounds` must be convertible into an instance of "
107 "`scipy.optimize.Bounds`.")
108 try:
109 bounds = Bounds(*bounds)
110 except TypeError as exc:
111 raise ValueError(message) from exc
112  
113 try:
114 lb = np.broadcast_to(bounds.lb, c.shape).astype(np.float64)
115 ub = np.broadcast_to(bounds.ub, c.shape).astype(np.float64)
116 except (ValueError, TypeError) as exc:
117 message = ("`bounds.lb` and `bounds.ub` must contain reals and "
118 "be broadcastable to `c.shape`.")
119 raise ValueError(message) from exc
120  
121 # constraints IV
122 if not constraints:
123 constraints = [LinearConstraint(np.empty((0, c.size)),
124 np.empty((0,)), np.empty((0,)))]
125 try:
126 A, b_l, b_u = _constraints_to_components(constraints)
127 except ValueError as exc:
128 message = ("`constraints` (or each element within `constraints`) must "
129 "be convertible into an instance of "
130 "`scipy.optimize.LinearConstraint`.")
131 raise ValueError(message) from exc
132  
133 if A.shape != (b_l.size, c.size):
134 message = "The shape of `A` must be (len(b_l), len(c))."
135 raise ValueError(message)
136 indptr, indices, data = A.indptr, A.indices, A.data.astype(np.float64)
137  
138 # options IV
139 options = options or {}
140 supported_options = {'disp', 'presolve', 'time_limit', 'node_limit',
141 'mip_rel_gap'}
142 unsupported_options = set(options).difference(supported_options)
143 if unsupported_options:
144 message = (f"Unrecognized options detected: {unsupported_options}. "
145 "These will be passed to HiGHS verbatim.")
146 warnings.warn(message, RuntimeWarning, stacklevel=3)
147 options_iv = {'log_to_console': options.pop("disp", False),
148 'mip_max_nodes': options.pop("node_limit", None)}
149 options_iv.update(options)
150  
151 return c, integrality, lb, ub, indptr, indices, data, b_l, b_u, options_iv

154 def milp(c, *, integrality=None, bounds=None, constraints=None, options=None):
155 r"""
156 Mixed-integer linear programming
157  
158 Solves problems of the following form:
159  
160 .. math::
161  
162 \min_x \ & c^T x \\
163 \mbox{such that} \ & b_l \leq A x \leq b_u,\\
164 & l \leq x \leq u, \\
165 & x_i \in \mathbb{Z}, i \in X_i
166  
167 where :math:`x` is a vector of decision variables;
168 :math:`c`, :math:`b_l`, :math:`b_u`, :math:`l`, and :math:`u` are vectors;
169 :math:`A` is a matrix, and :math:`X_i` is the set of indices of
170 decision variables that must be integral. (In this context, a
171 variable that can assume only integer values is said to be "integral";
172 it has an "integrality" constraint.)
173  
174 Alternatively, that's:
175  
176 minimize::
177  
178 c @ x
179  
180 such that::
181  
182 b_l <= A @ x <= b_u
183 l <= x <= u
184 Specified elements of x must be integers
185  
186 By default, ``l = 0`` and ``u = np.inf`` unless specified with
187 ``bounds``.
188  
189 Parameters
190 ----------
191 c : 1D dense array_like
192 The coefficients of the linear objective function to be minimized.
193 `c` is converted to a double precision array before the problem is
194 solved.
195 integrality : 1D dense array_like, optional
196 Indicates the type of integrality constraint on each decision variable.
197  
198 ``0`` : Continuous variable; no integrality constraint.
199  
200 ``1`` : Integer variable; decision variable must be an integer
201 within `bounds`.
202  
203 ``2`` : Semi-continuous variable; decision variable must be within
204 `bounds` or take value ``0``.
205  
206 ``3`` : Semi-integer variable; decision variable must be an integer
207 within `bounds` or take value ``0``.
208  
209 By default, all variables are continuous. `integrality` is converted
210 to an array of integers before the problem is solved.
211  
212 bounds : scipy.optimize.Bounds, optional
213 Bounds on the decision variables. Lower and upper bounds are converted
214 to double precision arrays before the problem is solved. The
215 ``keep_feasible`` parameter of the `Bounds` object is ignored. If
216 not specified, all decision variables are constrained to be
217 non-negative.
218 constraints : sequence of scipy.optimize.LinearConstraint, optional
219 Linear constraints of the optimization problem. Arguments may be
220 one of the following:
221  
222 1. A single `LinearConstraint` object
223 2. A single tuple that can be converted to a `LinearConstraint` object
224 as ``LinearConstraint(*constraints)``
225 3. A sequence composed entirely of objects of type 1. and 2.
226  
227 Before the problem is solved, all values are converted to double
228 precision, and the matrices of constraint coefficients are converted to
229 instances of `scipy.sparse.csc_array`. The ``keep_feasible`` parameter
230 of `LinearConstraint` objects is ignored.
231 options : dict, optional
232 A dictionary of solver options. The following keys are recognized.
233  
234 disp : bool (default: ``False``)
235 Set to ``True`` if indicators of optimization status are to be
236 printed to the console during optimization.
237 node_limit : int, optional
238 The maximum number of nodes (linear program relaxations) to solve
239 before stopping. Default is no maximum number of nodes.
240 presolve : bool (default: ``True``)
241 Presolve attempts to identify trivial infeasibilities,
242 identify trivial unboundedness, and simplify the problem before
243 sending it to the main solver.
244 time_limit : float, optional
245 The maximum number of seconds allotted to solve the problem.
246 Default is no time limit.
247 mip_rel_gap : float, optional
248 Termination criterion for MIP solver: solver will terminate when
249 the gap between the primal objective value and the dual objective
250 bound, scaled by the primal objective value, is <= mip_rel_gap.
251  
252 Returns
253 -------
254 res : OptimizeResult
255 An instance of :class:`scipy.optimize.OptimizeResult`. The object
256 is guaranteed to have the following attributes.
257  
258 status : int
259 An integer representing the exit status of the algorithm.
260  
261 ``0`` : Optimal solution found.
262  
263 ``1`` : Iteration or time limit reached.
264  
265 ``2`` : Problem is infeasible.
266  
267 ``3`` : Problem is unbounded.
268  
269 ``4`` : Other; see message for details.
270  
271 success : bool
272 ``True`` when an optimal solution is found and ``False`` otherwise.
273  
274 message : str
275 A string descriptor of the exit status of the algorithm.
276  
277 The following attributes will also be present, but the values may be
278 ``None``, depending on the solution status.
279  
280 x : ndarray
281 The values of the decision variables that minimize the
282 objective function while satisfying the constraints.
283 fun : float
284 The optimal value of the objective function ``c @ x``.
285 mip_node_count : int
286 The number of subproblems or "nodes" solved by the MILP solver.
287 mip_dual_bound : float
288 The MILP solver's final estimate of the lower bound on the optimal
289 solution.
290 mip_gap : float
291 The difference between the primal objective value and the dual
292 objective bound, scaled by the primal objective value.
293  
294 Notes
295 -----
296 `milp` is a wrapper of the HiGHS linear optimization software [1]_. The
297 algorithm is deterministic, and it typically finds the global optimum of
298 moderately challenging mixed-integer linear programs (when it exists).
299  
300 References
301 ----------
302 .. [1] Huangfu, Q., Galabova, I., Feldmeier, M., and Hall, J. A. J.
303 "HiGHS - high performance software for linear optimization."
304 https://highs.dev/
305 .. [2] Huangfu, Q. and Hall, J. A. J. "Parallelizing the dual revised
306 simplex method." Mathematical Programming Computation, 10 (1),
307 119-142, 2018. DOI: 10.1007/s12532-017-0130-5
308  
309 Examples
310 --------
311 Consider the problem at
312 https://en.wikipedia.org/wiki/Integer_programming#Example, which is
313 expressed as a maximization problem of two variables. Since `milp` requires
314 that the problem be expressed as a minimization problem, the objective
315 function coefficients on the decision variables are:
316  
317 >>> import numpy as np
318 >>> c = -np.array([0, 1])
319  
320 Note the negative sign: we maximize the original objective function
321 by minimizing the negative of the objective function.
322  
323 We collect the coefficients of the constraints into arrays like:
324  
325 >>> A = np.array([[-1, 1], [3, 2], [2, 3]])
326 >>> b_u = np.array([1, 12, 12])
327 >>> b_l = np.full_like(b_u, -np.inf, dtype=float)
328  
329 Because there is no lower limit on these constraints, we have defined a
330 variable ``b_l`` full of values representing negative infinity. This may
331 be unfamiliar to users of `scipy.optimize.linprog`, which only accepts
332 "less than" (or "upper bound") inequality constraints of the form
333 ``A_ub @ x <= b_u``. By accepting both ``b_l`` and ``b_u`` of constraints
334 ``b_l <= A_ub @ x <= b_u``, `milp` makes it easy to specify "greater than"
335 inequality constraints, "less than" inequality constraints, and equality
336 constraints concisely.
337  
338 These arrays are collected into a single `LinearConstraint` object like:
339  
340 >>> from scipy.optimize import LinearConstraint
341 >>> constraints = LinearConstraint(A, b_l, b_u)
342  
343 The non-negativity bounds on the decision variables are enforced by
344 default, so we do not need to provide an argument for `bounds`.
345  
346 Finally, the problem states that both decision variables must be integers:
347  
348 >>> integrality = np.ones_like(c)
349  
350 We solve the problem like:
351  
352 >>> from scipy.optimize import milp
353 >>> res = milp(c=c, constraints=constraints, integrality=integrality)
354 >>> res.x
355 [2.0, 2.0]
356  
357 Note that had we solved the relaxed problem (without integrality
358 constraints):
359  
360 >>> res = milp(c=c, constraints=constraints) # OR:
361 >>> # from scipy.optimize import linprog; res = linprog(c, A, b_u)
362 >>> res.x
363 [1.8, 2.8]
364  
365 we would not have obtained the correct solution by rounding to the nearest
366 integers.
367  
368 Other examples are given :ref:`in the tutorial <tutorial-optimize_milp>`.
369  
370 """
371 args_iv = _milp_iv(c, integrality, bounds, constraints, options)
372 c, integrality, lb, ub, indptr, indices, data, b_l, b_u, options = args_iv
373  
374 highs_res = _highs_wrapper(c, indptr, indices, data, b_l, b_u,
375 lb, ub, integrality, options)
376  
377 res = {}
378  
379 # Convert to scipy-style status and message
380 highs_status = highs_res.get('status', None)
381 highs_message = highs_res.get('message', None)
382 status, message = _highs_to_scipy_status_message(highs_status,
383 highs_message)
384 res['status'] = status
385 res['message'] = message
386 res['success'] = (status == 0)
387 x = highs_res.get('x', None)
388 res['x'] = np.array(x) if x is not None else None
389 res['fun'] = highs_res.get('fun', None)
390 res['mip_node_count'] = highs_res.get('mip_node_count', None)
391 res['mip_dual_bound'] = highs_res.get('mip_dual_bound', None)
392 res['mip_gap'] = highs_res.get('mip_gap', None)
393  
394 return OptimizeResult(res)
