16 def random_entry(n, min_eig, max_eig, case):
17  
18 # Generate random matrix
19 rand = np.random.uniform(-1, 1, (n, n))
20  
21 # QR decomposition
22 Q, _, _ = qr(rand, pivoting='True')
23  
24 # Generate random eigenvalues
25 eigvalues = np.random.uniform(min_eig, max_eig, n)
26 eigvalues = np.sort(eigvalues)[::-1]
27  
28 # Generate matrix
29 Qaux = np.multiply(eigvalues, Q)
30 A = np.dot(Qaux, Q.T)
31  
32 # Generate gradient vector accordingly
33 # to the case is being tested.
34 if case == 'hard':
35 g = np.zeros(n)
36 g[:-1] = np.random.uniform(-1, 1, n-1)
37 g = np.dot(Q, g)
38 elif case == 'jac_equal_zero':
39 g = np.zeros(n)
40 else:
41 g = np.random.uniform(-1, 1, n)
42  
43 return A, g

50 # Ill-conditioned triangular matrix
51 C = np.array([[1, 2, 3, 4],

54 [0, 0, 0, 10]])
55  
56 # Get svd decomposition

58  
59 # Get smallest singular value and correspondent right singular vector.
60 smin_svd = s[-1]

62  
63 # Estimate smallest singular value

275 @pytest.mark.fail_slow(10)
276 def test_for_random_entries(self):
277 # Seed
278 np.random.seed(1)
279  
280 # Dimension
281 n = 5
282  
283 for case in ('easy', 'hard', 'jac_equal_zero'):
284  
285 eig_limits = [(-20, -15),
286 (-10, -5),
287 (-10, 0),
288 (-5, 5),
289 (-10, 10),
290 (0, 10),
291 (5, 10),
292 (15, 20)]
293  
294 for min_eig, max_eig in eig_limits:
295 # Generate random symmetric matrix H with
296 # eigenvalues between min_eig and max_eig.
297 H, g = random_entry(n, min_eig, max_eig, case)
298  
299 # Trust radius
300 trust_radius_list = [0.1, 0.3, 0.6, 0.8, 1, 1.2, 3.3, 5.5, 10]
301  
302 for trust_radius in trust_radius_list:
303 # Solve subproblem with very high accuracy
304 subprob_ac = IterativeSubproblem(0,
305 lambda x: 0,
306 lambda x: g,
307 lambda x: H,
308 k_easy=1e-10,
309 k_hard=1e-10)
310  
311 p_ac, hits_boundary_ac = subprob_ac.solve(trust_radius)
312  
313 # Compute objective function value
314 J_ac = 1/2*np.dot(p_ac, np.dot(H, p_ac))+np.dot(g, p_ac)
315  
316 stop_criteria = [(0.1, 2),
317 (0.5, 1.1),
318 (0.9, 1.01)]
319  
320 for k_opt, k_trf in stop_criteria:
321  
322 # k_easy and k_hard computed in function
323 # of k_opt and k_trf accordingly to
324 # Conn, A. R., Gould, N. I., & Toint, P. L. (2000).
325 # "Trust region methods". Siam. p. 197.
326 k_easy = min(k_trf-1,
327 1-np.sqrt(k_opt))
328 k_hard = 1-k_opt
329  
330 # Solve subproblem
331 subprob = IterativeSubproblem(0,
332 lambda x: 0,
333 lambda x: g,
334 lambda x: H,
335 k_easy=k_easy,
336 k_hard=k_hard)
337 p, hits_boundary = subprob.solve(trust_radius)
338  
339 # Compute objective function value
340 J = 1/2*np.dot(p, np.dot(H, p))+np.dot(g, p)
341  
342 # Check if it respect k_trf
343 if hits_boundary:
344 assert_array_equal(np.abs(norm(p)-trust_radius) <=
345 (k_trf-1)*trust_radius, True)
346 else:
347 assert_equal(norm(p) <= trust_radius, True)
348  
349 # Check if it respect k_opt
350 assert_equal(J <= k_opt*J_ac, True)
