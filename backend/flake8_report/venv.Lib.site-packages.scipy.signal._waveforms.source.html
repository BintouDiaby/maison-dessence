16 def sawtooth(t, width=1):
17 """
18 Return a periodic sawtooth or triangle waveform.
19  
20 The sawtooth waveform has a period ``2*pi``, rises from -1 to 1 on the
21 interval 0 to ``width*2*pi``, then drops from 1 to -1 on the interval
22 ``width*2*pi`` to ``2*pi``. `width` must be in the interval [0, 1].
23  
24 Note that this is not band-limited. It produces an infinite number
25 of harmonics, which are aliased back and forth across the frequency
26 spectrum.
27  
28 Parameters
29 ----------
30 t : array_like
31 Time.
32 width : array_like, optional
33 Width of the rising ramp as a proportion of the total cycle.
34 Default is 1, producing a rising ramp, while 0 produces a falling
35 ramp. `width` = 0.5 produces a triangle wave.
36 If an array, causes wave shape to change over time, and must be the
37 same length as t.
38  
39 Returns
40 -------
41 y : ndarray
42 Output array containing the sawtooth waveform.
43  
44 Examples
45 --------
46 A 5 Hz waveform sampled at 500 Hz for 1 second:
47  
48 >>> import numpy as np
49 >>> from scipy import signal
50 >>> import matplotlib.pyplot as plt
51 >>> t = np.linspace(0, 1, 500)
52 >>> plt.plot(t, signal.sawtooth(2 * np.pi * 5 * t))
53  
54 """
55 t, w = asarray(t), asarray(width)
56 w = asarray(w + (t - t))
57 t = asarray(t + (w - w))
58 y = zeros(t.shape, dtype="d")
59  
60 # width must be between 0 and 1 inclusive
61 mask1 = (w > 1) | (w < 0)
62 place(y, mask1, nan)
63  
64 # take t modulo 2*pi
65 tmod = mod(t, 2 * pi)
66  
67 # on the interval 0 to width*2*pi function is
68 # tmod / (pi*w) - 1
69 mask2 = (1 - mask1) & (tmod < w * 2 * pi)
70 tsub = extract(mask2, tmod)
71 wsub = extract(mask2, w)
72 place(y, mask2, tsub / (pi * wsub) - 1)
73  
74 # on the interval width*2*pi to 2*pi function is
75 # (pi*(w+1)-tmod) / (pi*(1-w))
76  
77 mask3 = (1 - mask1) & (1 - mask2)
78 tsub = extract(mask3, tmod)
79 wsub = extract(mask3, w)
80 place(y, mask3, (pi * (wsub + 1) - tsub) / (pi * (1 - wsub)))
81 return y

84 def square(t, duty=0.5):
85 """
86 Return a periodic square-wave waveform.
87  
88 The square wave has a period ``2*pi``, has value +1 from 0 to
89 ``2*pi*duty`` and -1 from ``2*pi*duty`` to ``2*pi``. `duty` must be in
90 the interval [0,1].
91  
92 Note that this is not band-limited. It produces an infinite number
93 of harmonics, which are aliased back and forth across the frequency
94 spectrum.
95  
96 Parameters
97 ----------
98 t : array_like
99 The input time array.
100 duty : array_like, optional
101 Duty cycle. Default is 0.5 (50% duty cycle).
102 If an array, causes wave shape to change over time, and must be the
103 same length as t.
104  
105 Returns
106 -------
107 y : ndarray
108 Output array containing the square waveform.
109  
110 Examples
111 --------
112 A 5 Hz waveform sampled at 500 Hz for 1 second:
113  
114 >>> import numpy as np
115 >>> from scipy import signal
116 >>> import matplotlib.pyplot as plt
117 >>> t = np.linspace(0, 1, 500, endpoint=False)
118 >>> plt.plot(t, signal.square(2 * np.pi * 5 * t))
119 >>> plt.ylim(-2, 2)
120  
121 A pulse-width modulated sine wave:
122  
123 >>> plt.figure()
124 >>> sig = np.sin(2 * np.pi * t)
125 >>> pwm = signal.square(2 * np.pi * 30 * t, duty=(sig + 1)/2)
126 >>> plt.subplot(2, 1, 1)
127 >>> plt.plot(t, sig)
128 >>> plt.subplot(2, 1, 2)
129 >>> plt.plot(t, pwm)
130 >>> plt.ylim(-1.5, 1.5)
131  
132 """
133 t, w = asarray(t), asarray(duty)
134 w = asarray(w + (t - t))
135 t = asarray(t + (w - w))
136 y = zeros(t.shape, dtype="d")
137  
138 # width must be between 0 and 1 inclusive
139 mask1 = (w > 1) | (w < 0)
140 place(y, mask1, nan)
141  
142 # on the interval 0 to duty*2*pi function is 1
143 tmod = mod(t, 2 * pi)
144 mask2 = (1 - mask1) & (tmod < w * 2 * pi)
145 place(y, mask2, 1)
146  
147 # on the interval duty*2*pi to 2*pi function is
148 # (pi*(w+1)-tmod) / (pi*(1-w))
149 mask3 = (1 - mask1) & (1 - mask2)
150 place(y, mask3, -1)
151 return y

154 def gausspulse(t, fc=1000, bw=0.5, bwr=-6, tpr=-60, retquad=False,
155 retenv=False):
156 """
157 Return a Gaussian modulated sinusoid:
158  
159 ``exp(-a t^2) exp(1j*2*pi*fc*t).``
160  
161 If `retquad` is True, then return the real and imaginary parts
162 (in-phase and quadrature).
163 If `retenv` is True, then return the envelope (unmodulated signal).
164 Otherwise, return the real part of the modulated sinusoid.
165  
166 Parameters
167 ----------
168 t : ndarray or the string 'cutoff'
169 Input array.
170 fc : float, optional
171 Center frequency (e.g. Hz). Default is 1000.
172 bw : float, optional
173 Fractional bandwidth in frequency domain of pulse (e.g. Hz).
174 Default is 0.5.
175 bwr : float, optional
176 Reference level at which fractional bandwidth is calculated (dB).
177 Default is -6.
178 tpr : float, optional
179 If `t` is 'cutoff', then the function returns the cutoff
180 time for when the pulse amplitude falls below `tpr` (in dB).
181 Default is -60.
182 retquad : bool, optional
183 If True, return the quadrature (imaginary) as well as the real part
184 of the signal. Default is False.
185 retenv : bool, optional
186 If True, return the envelope of the signal. Default is False.
187  
188 Returns
189 -------
190 yI : ndarray
191 Real part of signal. Always returned.
192 yQ : ndarray
193 Imaginary part of signal. Only returned if `retquad` is True.
194 yenv : ndarray
195 Envelope of signal. Only returned if `retenv` is True.
196  
197 Examples
198 --------
199 Plot real component, imaginary component, and envelope for a 5 Hz pulse,
200 sampled at 100 Hz for 2 seconds:
201  
202 >>> import numpy as np
203 >>> from scipy import signal
204 >>> import matplotlib.pyplot as plt
205 >>> t = np.linspace(-1, 1, 2 * 100, endpoint=False)
206 >>> i, q, e = signal.gausspulse(t, fc=5, retquad=True, retenv=True)
207 >>> plt.plot(t, i, t, q, t, e, '--')
208  
209 """
210 if fc < 0:
211 raise ValueError(f"Center frequency (fc={fc:.2f}) must be >=0.")
212 if bw <= 0:
213 raise ValueError(f"Fractional bandwidth (bw={bw:.2f}) must be > 0.")
214 if bwr >= 0:
215 raise ValueError(f"Reference level for bandwidth (bwr={bwr:.2f}) "
216 "must be < 0 dB")
217  
218 # exp(-a t^2) <-> sqrt(pi/a) exp(-pi^2/a * f^2) = g(f)
219  
220 ref = pow(10.0, bwr / 20.0)
221 # fdel = fc*bw/2: g(fdel) = ref --- solve this for a
222 #
223 # pi^2/a * fc^2 * bw^2 /4=-log(ref)
224 a = -(pi * fc * bw) ** 2 / (4.0 * log(ref))
225  
226 if isinstance(t, str):
227 if t == 'cutoff': # compute cut_off point
228 # Solve exp(-a tc**2) = tref for tc
229 # tc = sqrt(-log(tref) / a) where tref = 10^(tpr/20)
230 if tpr >= 0:
231 raise ValueError("Reference level for time cutoff must "
232 "be < 0 dB")
233 tref = pow(10.0, tpr / 20.0)
234 return sqrt(-log(tref) / a)
235 else:
236 raise ValueError("If `t` is a string, it must be 'cutoff'")
237  
238 yenv = exp(-a * t * t)
239 yI = yenv * cos(2 * pi * fc * t)
240 yQ = yenv * sin(2 * pi * fc * t)
241 if not retquad and not retenv:
242 return yI
243 if not retquad and retenv:
244 return yI, yenv
245 if retquad and not retenv:
246 return yI, yQ
247 if retquad and retenv:
248 return yI, yQ, yenv

305 1. Parameter `method` in ``('linear', 'lin', 'li')``:

320 f_1 + \beta\, (t_1 - t)^2 & \text{otherwise,}
321 \end{cases}

323 \beta = \frac{f_1 - f_0}{t_1^2}

326 :math:`(t_1, f_1)`. By default, the vertex of the parabola is at

421 def _chirp_phase(t, f0, t1, f1, method='linear', vertex_zero=True):
422 """
423 Calculate the phase used by `chirp` to generate its output.
424  
425 See `chirp` for a description of the arguments.
426  
427 """
428 t = asarray(t)
429 f0 = float(f0)
430 t1 = float(t1)
431 f1 = float(f1)
432 if method in ['linear', 'lin', 'li']:
433 beta = (f1 - f0) / t1
434 phase = 2 * pi * (f0 * t + 0.5 * beta * t * t)
435  
436 elif method in ['quadratic', 'quad', 'q']:
437 beta = (f1 - f0) / (t1 ** 2)
438 if vertex_zero:
439 phase = 2 * pi * (f0 * t + beta * t ** 3 / 3)
440 else:
441 phase = 2 * pi * (f1 * t + beta * ((t1 - t) ** 3 - t1 ** 3) / 3)
442  
443 elif method in ['logarithmic', 'log', 'lo']:
444 if f0 * f1 <= 0.0:
445 raise ValueError("For a logarithmic chirp, f0 and f1 must be "
446 "nonzero and have the same sign.")
447 if f0 == f1:
448 phase = 2 * pi * f0 * t
449 else:
450 beta = t1 / log(f1 / f0)
451 phase = 2 * pi * beta * f0 * (pow(f1 / f0, t / t1) - 1.0)
452  
453 elif method in ['hyperbolic', 'hyp']:
454 if f0 == 0 or f1 == 0:
455 raise ValueError("For a hyperbolic chirp, f0 and f1 must be "
456 "nonzero.")
457 if f0 == f1:
458 # Degenerate case: constant frequency.
459 phase = 2 * pi * f0 * t
460 else:
461 # Singular point: the instantaneous frequency blows up
462 # when t == sing.
463 sing = -f1 * t1 / (f0 - f1)
464 phase = 2 * pi * (-sing * f0) * log(np.abs(1 - t/sing))
465  
466 else:
  • E501 Line too long (80 > 79 characters)
467 raise ValueError("method must be 'linear', 'quadratic', 'logarithmic', "
  • E501 Line too long (82 > 79 characters)
468 f"or 'hyperbolic', but a value of {method!r} was given.")
469  
470 return phase

473 def sweep_poly(t, poly, phi=0):
474 """
475 Frequency-swept cosine generator, with a time-dependent frequency.
476  
477 This function generates a sinusoidal function whose instantaneous
478 frequency varies with time. The frequency at time `t` is given by
479 the polynomial `poly`.
480  
481 Parameters
482 ----------
483 t : ndarray
484 Times at which to evaluate the waveform.
485 poly : 1-D array_like or instance of numpy.poly1d
486 The desired frequency expressed as a polynomial. If `poly` is
487 a list or ndarray of length n, then the elements of `poly` are
488 the coefficients of the polynomial, and the instantaneous
489 frequency is
490  
491 ``f(t) = poly[0]*t**(n-1) + poly[1]*t**(n-2) + ... + poly[n-1]``
492  
493 If `poly` is an instance of numpy.poly1d, then the
494 instantaneous frequency is
495  
496 ``f(t) = poly(t)``
497  
498 phi : float, optional
499 Phase offset, in degrees, Default: 0.
500  
501 Returns
502 -------
503 sweep_poly : ndarray
504 A numpy array containing the signal evaluated at `t` with the
505 requested time-varying frequency. More precisely, the function
506 returns ``cos(phase + (pi/180)*phi)``, where `phase` is the integral
507 (from 0 to t) of ``2 * pi * f(t)``; ``f(t)`` is defined above.
508  
509 See Also
510 --------
511 chirp
512  
513 Notes
514 -----
515 .. versionadded:: 0.8.0
516  
517 If `poly` is a list or ndarray of length `n`, then the elements of
518 `poly` are the coefficients of the polynomial, and the instantaneous
519 frequency is:
520  
521 ``f(t) = poly[0]*t**(n-1) + poly[1]*t**(n-2) + ... + poly[n-1]``
522  
523 If `poly` is an instance of `numpy.poly1d`, then the instantaneous
524 frequency is:
525  
526 ``f(t) = poly(t)``
527  
528 Finally, the output `s` is:
529  
530 ``cos(phase + (pi/180)*phi)``
531  
532 where `phase` is the integral from 0 to `t` of ``2 * pi * f(t)``,
533 ``f(t)`` as defined above.
534  
535 Examples
536 --------
537 Compute the waveform with instantaneous frequency::
538  
539 f(t) = 0.025*t**3 - 0.36*t**2 + 1.25*t + 2
540  
541 over the interval 0 <= t <= 10.
542  
543 >>> import numpy as np
544 >>> from scipy.signal import sweep_poly
545 >>> p = np.poly1d([0.025, -0.36, 1.25, 2.0])
546 >>> t = np.linspace(0, 10, 5001)
547 >>> w = sweep_poly(t, p)
548  
549 Plot it:
550  
551 >>> import matplotlib.pyplot as plt
552 >>> plt.subplot(2, 1, 1)
553 >>> plt.plot(t, w)
554 >>> plt.title("Sweep Poly\\nwith frequency " +
555 ... "$f(t) = 0.025t^3 - 0.36t^2 + 1.25t + 2$")
556 >>> plt.subplot(2, 1, 2)
557 >>> plt.plot(t, p(t), 'r', label='f(t)')
558 >>> plt.legend()
559 >>> plt.xlabel('t')
560 >>> plt.tight_layout()
561 >>> plt.show()
562  
563 """
564 # 'phase' is computed in _sweep_poly_phase, to make testing easier.
565 phase = _sweep_poly_phase(t, poly)
566 # Convert to radians.
567 phi *= pi / 180
568 return cos(phase + phi)

584 def unit_impulse(shape, idx=None, dtype=float):
585 r"""
586 Unit impulse signal (discrete delta function) or unit basis vector.
587  
588 Parameters
589 ----------
590 shape : int or tuple of int
591 Number of samples in the output (1-D), or a tuple that represents the
592 shape of the output (N-D).
593 idx : None or int or tuple of int or 'mid', optional
594 Index at which the value is 1. If None, defaults to the 0th element.
595 If ``idx='mid'``, the impulse will be centered at ``shape // 2`` in
596 all dimensions. If an int, the impulse will be at `idx` in all
597 dimensions.
598 dtype : data-type, optional
599 The desired data-type for the array, e.g., ``numpy.int8``. Default is
600 ``numpy.float64``.
601  
602 Returns
603 -------
604 y : ndarray
605 Output array containing an impulse signal.
606  
607 Notes
608 -----
609 In digital signal processing literature the unit impulse signal is often
610 represented by the Kronecker delta. [1]_ I.e., a signal :math:`u_k[n]`,
