2  
3 This module provides the functions stft_compare() and istft_compare(), which,
4 compares the output between the existing (i)stft() and the shortTimeFFT based

6  
7 For testing add the following imports to the file ``tests/test_spectral.py``::

33 def _stft_wrapper(x, fs=1.0, window='hann', nperseg=256, noverlap=None,
34 nfft=None, detrend=False, return_onesided=True,
35 boundary='zeros', padded=True, axis=-1, scaling='spectrum'):
36 """Wrapper for the SciPy `stft()` function based on `ShortTimeFFT` for
37 unit testing.
38  
39 Handling the boundary and padding is where `ShortTimeFFT` and `stft()`
40 differ in behavior. Parts of `_spectral_helper()` were copied to mimic
41 the` stft()` behavior.
42  
43 This function is meant to be solely used by `stft_compare()`.
44 """

46 raise ValueError(f"Parameter {scaling=} not in ['spectrum', 'psd']!")

51 'constant': const_ext,
52 'zeros': zero_ext,
53 None: None}

56 raise ValueError(f"Unknown boundary option '{boundary}', must be one" +
57 f" of: {list(boundary_funcs.keys())}")

75 nfft = int(nfft)

86 # Padding occurs after boundary extension, so that the extended signal ends
87 # in zeros, instead of introducing an impulse at the end.
88 # I.e. if x = [..., 3, 2]
89 # extend then pad -> [..., 3, 2, 2, 3, 0, 0, 0]
90 # pad then extend -> [..., 3, 2, 0, 0, 0, 2, 3]
91  
92 if boundary is not None:
93 ext_func = boundary_funcs[boundary]
94 # Extend by nperseg//2 in front and back:
95 x = ext_func(x, nperseg//2, axis=axis)
96  
97 if padded:
98 # Pad to integer number of windowed segments
99 # I.e make x.shape[-1] = nperseg + (nseg-1)*nstep, with integer nseg
100 x = np.moveaxis(x, axis, -1)
101  
102 # This is an edge case where shortTimeFFT returns one more time slice
103 # than the Scipy stft() shorten to remove last time slice:
104 if n % 2 == 1 and nperseg % 2 == 1 and noverlap % 2 == 1:
105 x = x[..., : -1]
106  
107 nadd = (-(x.shape[-1]-nperseg) % nstep) % nperseg
108 zeros_shape = list(x.shape[:-1]) + [nadd]
109 x = np.concatenate((x, np.zeros(zeros_shape)), axis=-1)
110 x = np.moveaxis(x, -1, axis)
111  
112 # ... end original _spectral_helper() code.
113 scale_to = {'spectrum': 'magnitude', 'psd': 'psd'}[scaling]
114  
115 if np.iscomplexobj(x) and return_onesided:
116 return_onesided = False
117 # using cast() to make mypy happy:
  • E501 Line too long (81 > 79 characters)
118 fft_mode = cast(FFT_MODE_TYPE, 'onesided' if return_onesided else 'twosided')
119  
120 ST = ShortTimeFFT(win, nstep, fs, fft_mode=fft_mode, mfft=nfft,
121 scale_to=scale_to, phase_shift=None)
122  
123 k_off = nperseg // 2
124 p0 = 0 # ST.lower_border_end[1] + 1
125 nn = x.shape[axis] if padded else n+k_off+1
126 # number of frames akin to legacy stft computation
  • W291 Trailing whitespace
127 p1 = (x.shape[axis] - nperseg) // nstep + 1
128  
129 detr = None if detrend is False else detrend
130 Sxx = ST.stft_detrend(x, detr, p0, p1, k_offset=k_off, axis=axis)
131 t = ST.t(nn, 0, p1 - p0, k_offset=0 if boundary is not None else k_off)
132 if x.dtype in (np.float32, np.complex64):
133 Sxx = Sxx.astype(np.complex64)
134  
135 return ST.f, t, Sxx

138 def _istft_wrapper(Zxx, fs=1.0, window='hann', nperseg=None, noverlap=None,
139 nfft=None, input_onesided=True, boundary=True, time_axis=-1,
140 freq_axis=-2, scaling='spectrum') -> \
141 tuple[np.ndarray, np.ndarray, tuple[int, int]]:
142 """Wrapper for the SciPy `istft()` function based on `ShortTimeFFT` for
143 unit testing.
144  
145 Note that only option handling is implemented as far as to handle the unit
146 tests. E.g., the case ``nperseg=None`` is not handled.
147  
148 This function is meant to be solely used by `istft_compare()`.
149 """
150 # *** Lines are taken from _spectral_py.istft() ***:
151 if Zxx.ndim < 2:
152 raise ValueError('Input stft must be at least 2d!')
153  
154 if freq_axis == time_axis:
155 raise ValueError('Must specify differing time and frequency axes!')
156  
157 nseg = Zxx.shape[time_axis]
158  
159 if input_onesided:
160 # Assume even segment length
161 n_default = 2*(Zxx.shape[freq_axis] - 1)
162 else:
163 n_default = Zxx.shape[freq_axis]
164  
165 # Check windowing parameters
166 if nperseg is None:
167 nperseg = n_default
168 else:
169 nperseg = int(nperseg)
170 if nperseg < 1:
171 raise ValueError('nperseg must be a positive integer')
172  
173 if nfft is None:
174 if input_onesided and (nperseg == n_default + 1):
175 # Odd nperseg, no FFT padding
176 nfft = nperseg
177 else:
178 nfft = n_default
179 elif nfft < nperseg:
180 raise ValueError('nfft must be greater than or equal to nperseg.')
181 else:
182 nfft = int(nfft)
183  
184 if noverlap is None:
185 noverlap = nperseg//2
186 else:
187 noverlap = int(noverlap)
188 if noverlap >= nperseg:
189 raise ValueError('noverlap must be less than nperseg.')
190 nstep = nperseg - noverlap
191  
192 # Get window as array
193 if isinstance(window, str) or type(window) is tuple:
194 win = get_window(window, nperseg)
195 else:
196 win = np.asarray(window)
197 if len(win.shape) != 1:
198 raise ValueError('window must be 1-D')
199 if win.shape[0] != nperseg:
200 raise ValueError(f'window must have length of {nperseg}')
201  
202 outputlength = nperseg + (nseg-1)*nstep
203 # *** End block of: Taken from _spectral_py.istft() ***
204  
205 # Using cast() to make mypy happy:
  • E501 Line too long (80 > 79 characters)
206 fft_mode = cast(FFT_MODE_TYPE, 'onesided' if input_onesided else 'twosided')
207 scale_to = cast(Literal['magnitude', 'psd'],
208 {'spectrum': 'magnitude', 'psd': 'psd'}[scaling])
209  
210 ST = ShortTimeFFT(win, nstep, fs, fft_mode=fft_mode, mfft=nfft,
211 scale_to=scale_to, phase_shift=None)
212  
213 if boundary:
214 j = nperseg if nperseg % 2 == 0 else nperseg - 1
215 k0 = ST.k_min + nperseg // 2
216 k1 = outputlength - j + k0
217 else:
218 raise NotImplementedError("boundary=False does not make sense with" +
219 "ShortTimeFFT.istft()!")
220  
221 x = ST.istft(Zxx, k0=k0, k1=k1, f_axis=freq_axis, t_axis=time_axis)
222 t = np.arange(k1 - k0) * ST.T
223 k_hi = ST.upper_border_begin(k1 - k0)[0]
224 # using cast() to make mypy happy:
225 return t, x, (ST.lower_border_end[0], k_hi)

228 def stft_compare(x, fs=1.0, window='hann', nperseg=256, noverlap=None,
229 nfft=None, detrend=False, return_onesided=True,
230 boundary='zeros', padded=True, axis=-1, scaling='spectrum'):
231 """Assert that the results from the existing `stft()` and `_stft_wrapper()`
232 are close to each other.
233  
234 For comparing the STFT values an absolute tolerance of the floating point
235 resolution was added to circumvent problems with the following tests:
236 * For float32 the tolerances are much higher in
237 TestSTFT.test_roundtrip_float32()).
238 * The TestSTFT.test_roundtrip_scaling() has a high relative deviation.
239 Interestingly this did not appear in Scipy 1.9.1 but only in the current
240 development version.
241 """
242 kw = dict(x=x, fs=fs, window=window, nperseg=nperseg, noverlap=noverlap,
243 nfft=nfft, detrend=detrend, return_onesided=return_onesided,
244 boundary=boundary, padded=padded, axis=axis, scaling=scaling)
245 f, t, Zxx = stft(**kw)
246 f_wrapper, t_wrapper, Zxx_wrapper = _stft_wrapper(**kw)
247  
248 e_msg_part = " of `stft_wrapper()` differ from `stft()`."
249 assert_allclose(f_wrapper, f, err_msg=f"Frequencies {e_msg_part}")
250 assert_allclose(t_wrapper, t, err_msg=f"Time slices {e_msg_part}")
251  
252 # Adapted tolerances to account for:
253 atol = np.finfo(Zxx.dtype).resolution * 2
254 assert_allclose(Zxx_wrapper, Zxx, atol=atol,
255 err_msg=f"STFT values {e_msg_part}")
256 return f, t, Zxx

259 def istft_compare(Zxx, fs=1.0, window='hann', nperseg=None, noverlap=None,
260 nfft=None, input_onesided=True, boundary=True, time_axis=-1,
261 freq_axis=-2, scaling='spectrum'):
262 """Assert that the results from the existing `istft()` and
263 `_istft_wrapper()` are close to each other.
264  
265 Quirks:
266 * If ``boundary=False`` the comparison is skipped, since it does not
267 make sense with ShortTimeFFT.istft(). Only used in test
268 TestSTFT.test_roundtrip_boundary_extension().
269 * If ShortTimeFFT.istft() decides the STFT is not invertible, the
270 comparison is skipped, since istft() only emits a warning and does not
271 return a correct result. Only used in
272 ShortTimeFFT.test_roundtrip_not_nola().
273 * For comparing the signals an absolute tolerance of the floating point
274 resolution was added to account for the low accuracy of float32 (Occurs
275 only in TestSTFT.test_roundtrip_float32()).
276 """
277 kw = dict(Zxx=Zxx, fs=fs, window=window, nperseg=nperseg,
278 noverlap=noverlap, nfft=nfft, input_onesided=input_onesided,
279 boundary=boundary, time_axis=time_axis, freq_axis=freq_axis,
280 scaling=scaling)
281  
282 t, x = istft(**kw)
283 if not boundary: # skip test_roundtrip_boundary_extension():
284 return t, x # _istft_wrapper does() not implement this case
285 try: # if inversion fails, istft() only emits a warning:
286 t_wrapper, x_wrapper, (k_lo, k_hi) = _istft_wrapper(**kw)
287 except ValueError as v: # Do nothing if inversion fails:
288 if v.args[0] == "Short-time Fourier Transform not invertible!":
289 return t, x
290 raise v
291  
292 e_msg_part = " of `istft_wrapper()` differ from `istft()`"
293 assert_allclose(t, t_wrapper, err_msg=f"Sample times {e_msg_part}")
294  
295 # Adapted tolerances to account for resolution loss:
296 atol = np.finfo(x.dtype).resolution*2 # instead of default atol = 0
297 rtol = 1e-7 # default for np.allclose()
298  
299 # Relax atol on 32-Bit platforms a bit to pass CI tests.
300 # - Not clear why there are discrepancies (in the FFT maybe?)
301 # - Not sure what changed on 'i686' since earlier on those test passed
302 if x.dtype == np.float32 and platform.machine() == 'i686':
303 # float32 gets only used by TestSTFT.test_roundtrip_float32() so
304 # we are using the tolerances from there to circumvent CI problems
305 atol, rtol = 1e-4, 1e-5
306 elif platform.machine() in ('aarch64', 'i386', 'i686'):
307 atol = max(atol, 1e-12) # 2e-15 seems too tight for 32-Bit platforms
308  
309 assert_allclose(x_wrapper[k_lo:k_hi], x[k_lo:k_hi], atol=atol, rtol=rtol,
310 err_msg=f"Signal values {e_msg_part}")
311 return t, x
