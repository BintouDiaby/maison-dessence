43 def alt_sg_coeffs(window_length, polyorder, pos):
44 """This is an alternative implementation of the SG coefficients.
45  
46 It uses numpy.polyfit and numpy.polyval. The results should be
47 equivalent to those of savgol_coeffs(), but this implementation
48 is slower.
49  
50 window_length should be odd.
51  
52 """
53 if pos is None:
54 pos = window_length // 2
55 t = np.arange(window_length)
56 unit = (t == pos).astype(int)
57 h = np.polyval(np.polyfit(t, unit, polyorder), t)
58 return h

128 def test_sg_coeffs_deriv():
129 # The data in `x` is a sampled parabola, so using savgol_coeffs with an
130 # order 2 or higher polynomial should give exact results.
131 i = np.array([-2.0, 0.0, 2.0, 4.0, 6.0])

171 def test_sg_coeffs_even_window_length():
172 # Simple case - deriv=0, polyorder=0, 1
173 window_lengths = [4, 6, 8, 10, 12, 14, 16]
174 for length in window_lengths:

177  
178 # Verify with closed forms
179 # deriv=1, polyorder=1, 2
180 def h_p_d_closed_form_1(k, m):
181 return 6*(k - 0.5)/((2*m + 1)*m*(2*m - 1))
182  
183 # deriv=2, polyorder=2
184 def h_p_d_closed_form_2(k, m):
185 numer = 15*(-4*m**2 + 1 + 12*(k - 0.5)**2)
186 denom = 4*(2*m + 1)*(m + 1)*m*(m - 1)*(2*m - 1)
187 return numer/denom
188

190 m = length//2
191 expected_output = [h_p_d_closed_form_1(k, m)
