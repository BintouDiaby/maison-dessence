53 def upfirdn_naive(x, h, up=1, down=1):
54 """Naive upfirdn processing in Python.
55  
56 Note: arg order (x, h) differs to facilitate apply_along_axis use.
57 """
58 x = np.asarray(x)
59 h = np.asarray(h)
60 out = np.zeros(len(x) * up, x.dtype)
61 out[::up] = x
62 out = np.convolve(h, out)[::down][:_output_len(len(h), len(x), up, down)]
63 return out

68 def __init__(self, up, down, h, x_dtype):
69 self.up = up
70 self.down = down
71 self.h = np.atleast_1d(h)

83 x += 1j * self.rng.randn(10)

91 x += 1j * self.rng.randn(*size)
92 for axis in range(len(size)):

94 x = x[:, ::2, 1::3].T

98 def scrub(self, x, axis=-1):
99 yr = np.apply_along_axis(upfirdn_naive, axis, x,
100 self.h, self.up, self.down)
101 want_len = _output_len(len(self.h), x.shape[axis], self.up, self.down)
102 assert yr.shape[axis] == want_len
103 y = upfirdn(self.h, x, self.up, self.down, axis=axis)
104 assert y.shape[axis] == want_len
105 assert y.shape == yr.shape
106 dtypes = (self.h.dtype, x.dtype)
107 if all(d == np.complex64 for d in dtypes):
108 assert y.dtype == np.complex64
109 elif np.complex64 in dtypes and np.float32 in dtypes:
110 assert y.dtype == np.complex64
111 elif all(d == np.float32 for d in dtypes):
112 assert y.dtype == np.float32
113 elif np.complex128 in dtypes or np.complex64 in dtypes:
114 assert y.dtype == np.complex128
115 else:
116 assert y.dtype == np.float64
117 xp_assert_close(yr.astype(y.dtype), y)
