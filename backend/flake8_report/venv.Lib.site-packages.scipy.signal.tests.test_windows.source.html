157 def test_correctness(self, xp):
158 """This test ensures the correctness of the implemented Taylor
159 Windowing function. A Taylor Window of 1024 points is created, its FFT
160 is taken, and the Peak Sidelobe Level (PSLL) and 3dB and 18dB bandwidth
161 are found and checked.
162  
163 A publication from Sandia National Laboratories was used as reference
164 for the correctness values [1]_.
165  
166 References
167 -----
168 .. [1] Armin Doerry, "Catalog of Window Taper Functions for
169 Sidelobe Control", 2017.
170 https://www.researchgate.net/profile/Armin_Doerry/publication/316281181_Catalog_of_Window_Taper_Functions_for_Sidelobe_Control/links/58f92cb2a6fdccb121c9d54d/Catalog-of-Window-Taper-Functions-for-Sidelobe-Control.pdf
171 """
172 M_win = 1024
173 N_fft = 131072
174 # Set norm=False for correctness as the values obtained from the
175 # scientific publication do not normalize the values. Normalizing
176 # changes the sidelobe level from the desired value.
177 w = windows.taylor(M_win, nbar=4, sll=35, norm=False, sym=False, xp=xp)
178 f = fft(w, N_fft)
179  
180 f_np = np.asarray(f)
181 spec = 20 * np.log10(np.abs(f_np / np.max(f_np)))
182  
183 first_zero = np.argmax(np.diff(spec) > 0)
184  
185 PSLL = np.max(spec[first_zero:-first_zero])
186  
187 BW_3dB = 2*np.argmax(spec <= -3.0102999566398121) / N_fft * M_win
188 BW_18dB = 2*np.argmax(spec <= -18.061799739838872) / N_fft * M_win
189  
190 assert math.isclose(PSLL, -35.1672, abs_tol=1)
191 assert math.isclose(BW_3dB, 1.1822, abs_tol=0.1)
192 assert math.isclose(BW_18dB, 2.6112, abs_tol=0.1)
