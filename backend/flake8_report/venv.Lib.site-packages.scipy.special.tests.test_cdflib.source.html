150 def _binomial_cdf(k, n, p):
151 k, n, p = mpmath.mpf(k), mpmath.mpf(n), mpmath.mpf(p)
152 if k <= 0:
153 return mpmath.mpf(0)
154 elif k >= n:
155 return mpmath.mpf(1)
156  
157 onemp = mpmath.fsub(1, p, exact=True)
158 return mpmath.betainc(n - k, k + 1, x2=onemp, regularized=True)

161 def _f_cdf(dfn, dfd, x):
162 if x < 0:
163 return mpmath.mpf(0)
164 dfn, dfd, x = mpmath.mpf(dfn), mpmath.mpf(dfd), mpmath.mpf(x)
165 ub = dfn*x/(dfn*x + dfd)
166 res = mpmath.betainc(dfn/2, dfd/2, x2=ub, regularized=True)
167 return res

170 def _student_t_cdf(df, t, dps=None):
171 if dps is None:
172 dps = mpmath.mp.dps
173 with mpmath.workdps(dps):
174 df, t = mpmath.mpf(df), mpmath.mpf(t)
175 fac = mpmath.hyp2f1(0.5, 0.5*(df + 1), 1.5, -t**2/df)
176 fac *= t*mpmath.gamma(0.5*(df + 1))
177 fac /= mpmath.sqrt(mpmath.pi*df)*mpmath.gamma(0.5*df)
178 return 0.5 + fac

181 def _noncentral_chi_pdf(t, df, nc):
182 res = mpmath.besseli(df/2 - 1, mpmath.sqrt(nc*t))
183 res *= mpmath.exp(-(t + nc)/2)*(t/nc)**(df/4 - 1/2)/2
184 return res

188 if dps is None:
189 dps = mpmath.mp.dps
