46 def _factorize_int(n):
47 """Return a sorted list of the unique prime factors of a positive integer.
48 """
49 # NOTE: There are lots faster ways to do this, but this isn't terrible.
50 factors = set()
51 for p in primes_from_2_to(int(np.sqrt(n)) + 1):
52 while not (n % p):
53 factors.add(p)
54 n //= p
55 if n == 1:
56 break
57 if n != 1:
58 factors.add(n)
59 return sorted(factors)

62 def _primitive_root(p):
63 """Compute a primitive root of the prime number `p`.
64  
65 Used in the CBC lattice construction.
66  
67 References
68 ----------
69 .. [1] https://en.wikipedia.org/wiki/Primitive_root_modulo_n
70 """
71 # p is prime
72 pm = p - 1
73 factors = _factorize_int(pm)
74 n = len(factors)
75 r = 2
76 k = 0
77 while k < n:
78 d = pm // factors[k]
79 # pow() doesn't like numpy scalar types.
80 rd = pow(int(r), int(d), int(p))
81 if rd == 1:
82 r += 1
83 k = 0
84 else:
85 k += 1
86 return r

89 def _cbc_lattice(n_dim, n_qmc_samples):
90 """Compute a QMC lattice generator using a Fast CBC construction.
91  
92 Parameters
93 ----------
94 n_dim : int > 0
95 The number of dimensions for the lattice.
96 n_qmc_samples : int > 0
97 The desired number of QMC samples. This will be rounded down to the
98 nearest prime to enable the CBC construction.
99  
100 Returns
101 -------
102 q : float array : shape=(n_dim,)
103 The lattice generator vector. All values are in the open interval
104 ``(0, 1)``.
105 actual_n_qmc_samples : int
106 The prime number of QMC samples that must be used with this lattice,
107 no more, no less.
108  
109 References
110 ----------
111 .. [1] Nuyens, D. and Cools, R. "Fast Component-by-Component Construction,
112 a Reprise for Different Kernels", In H. Niederreiter and D. Talay,
113 editors, Monte-Carlo and Quasi-Monte Carlo Methods 2004,
114 Springer-Verlag, 2006, 371-385.
115 """
116 # Round down to the nearest prime number.
117 primes = primes_from_2_to(n_qmc_samples + 1)
118 n_qmc_samples = primes[-1]
119  
120 bt = np.ones(n_dim)
121 gm = np.hstack([1.0, 0.8 ** np.arange(n_dim - 1)])
122 q = 1
123 w = 0
124 z = np.arange(1, n_dim + 1)
125 m = (n_qmc_samples - 1) // 2
126 g = _primitive_root(n_qmc_samples)
127 # Slightly faster way to compute perm[j] = pow(g, j, n_qmc_samples)
128 # Shame that we don't have modulo pow() implemented as a ufunc.
129 perm = np.ones(m, dtype=int)
130 for j in range(m - 1):
131 perm[j + 1] = (g * perm[j]) % n_qmc_samples
132 perm = np.minimum(n_qmc_samples - perm, perm)
133 pn = perm / n_qmc_samples
134 c = pn * pn - pn + 1.0 / 6
135 fc = fft(c)
136 for s in range(1, n_dim):
137 reordered = np.hstack([
138 c[:w+1][::-1],
139 c[w+1:m][::-1],
140 ])
141 q = q * (bt[s-1] + gm[s-1] * reordered)
142 w = ifft(fc * fft(q)).real.argmin()
143 z[s] = perm[w]
144 q = z / n_qmc_samples
145 return q, n_qmc_samples

148 def _qauto(func, covar, low, high, rng, error=1e-3, limit=10_000, **kwds):
149 """Automatically rerun the integration to get the required error bound.
150  
151 Parameters
152 ----------
153 func : callable
154 Either :func:`_qmvn` or :func:`_qmvt`.
155 covar, low, high : array
156 As specified in :func:`_qmvn` and :func:`_qmvt`.
157 rng : Generator, optional
158 default_rng(), yada, yada
159 error : float > 0
160 The desired error bound.
161 limit : int > 0:
162 The rough limit of the number of integration points to consider. The
163 integration will stop looping once this limit has been *exceeded*.
164 **kwds :
165 Other keyword arguments to pass to `func`. When using :func:`_qmvt`, be
166 sure to include ``nu=`` as one of these.
167  
168 Returns
169 -------
170 prob : float
171 The estimated probability mass within the bounds.
172 est_error : float
173 3 times the standard error of the batch estimates.
174 n_samples : int
175 The number of integration points actually used.
176 """
177 n = len(covar)
178 n_samples = 0
179 if n == 1:
180 prob = phi(high / covar**0.5) - phi(low / covar**0.5)
181 # More or less
182 est_error = 1e-15
183 elif n == 2:
184 prob = _bvn(low, high, covar)
185 est_error = 1e-15
186 else:
187 mi = min(limit, n * 1000)
188 prob = 0.0
189 est_error = 1.0
190 ei = 0.0
191 while est_error > error and n_samples < limit:
192 mi = round(np.sqrt(2) * mi)
193 pi, ei, ni = func(mi, covar, low, high, rng=rng, **kwds)
194 n_samples += ni
195 wt = 1.0 / (1 + (ei / est_error)**2)
196 prob += wt * (pi - prob)
197 est_error = np.sqrt(wt) * ei
198 return prob, est_error, n_samples

203

206 m : int > n_batches
207 The number of points to sample. This number will be divided into
208 `n_batches` batches that apply random offsets of the sampling lattice

212 low, high : (n,) float array

218 n_batches : int > 0, optional
219 The number of QMC batches to apply.
220  
221 Returns

246 def _mvn_qmc_integrand(covar, low, high, use_tent=False):
247 """Transform the multivariate normal integration into a QMC integrand over
248 a unit hypercube.
249  
250 The dimensionality of the resulting hypercube integration domain is one
251 less than the dimensionality of the original integrand. Note that this
252 transformation subsumes the integration bounds in order to account for
253 infinite bounds. The QMC integration one does with the returned integrand
254 should be on the unit hypercube.
255  
256 Parameters
257 ----------
258 covar : (n, n) float array
259 Possibly singular, positive semidefinite symmetric covariance matrix.
260 low, high : (n,) float array
261 The low and high integration bounds.
262 use_tent : bool, optional
263 If True, then use tent periodization. Only helpful for lattice rules.
264  
265 Returns
266 -------
267 integrand : Callable[[NDArray], NDArray]
268 The QMC-integrable integrand. It takes an
269 ``(n_qmc_samples, ndim_integrand)`` array of QMC samples in the unit
270 hypercube and returns the ``(n_qmc_samples,)`` evaluations of at these
271 QMC points.
272 ndim_integrand : int
273 The dimensionality of the integrand. Equal to ``n-1``.
274 """
275 cho, lo, hi = _permuted_cholesky(covar, low, high)
276 n = cho.shape[0]
277 ndim_integrand = n - 1
278 ct = cho[0, 0]
279 c = phi(lo[0] / ct)
280 d = phi(hi[0] / ct)
281 ci = c
282 dci = d - ci
283  
284 def integrand(*zs):
285 ndim_qmc = len(zs)
286 n_qmc_samples = len(np.atleast_1d(zs[0]))
287 assert ndim_qmc == ndim_integrand
288 y = np.zeros((ndim_qmc, n_qmc_samples))
289 c = np.full(n_qmc_samples, ci)
290 dc = np.full(n_qmc_samples, dci)
291 pv = dc.copy()
292 for i in range(1, n):
293 if use_tent:
294 # Tent periodization transform.
295 x = abs(2 * zs[i-1] - 1)
296 else:
297 x = zs[i-1]
298 y[i - 1, :] = phinv(c + x * dc)
299 s = cho[i, :i] @ y[:i, :]
300 ct = cho[i, i]
301 c = phi((lo[i] - s) / ct)
302 d = phi((hi[i] - s) / ct)
303 dc = d - c
304 pv = pv * dc
305 return pv
306  
307 return integrand, ndim_integrand

310 def _qmvt(m, nu, covar, low, high, rng, lattice='cbc', n_batches=10):
311 """Multivariate t integration over box bounds.
312  
313 Parameters
314 ----------
315 m : int > n_batches
316 The number of points to sample. This number will be divided into
317 `n_batches` batches that apply random offsets of the sampling lattice
318 for each batch in order to estimate the error.
319 nu : float >= 0
320 The shape parameter of the multivariate t distribution.
321 covar : (n, n) float array
322 Possibly singular, positive semidefinite symmetric covariance matrix.
323 low, high : (n,) float array
324 The low and high integration bounds.
325 rng : Generator, optional
326 default_rng(), yada, yada
327 lattice : 'cbc' or callable
328 The type of lattice rule to use to construct the integration points.
329 n_batches : int > 0, optional
330 The number of QMC batches to apply.
331  
332 Returns
333 -------
334 prob : float
335 The estimated probability mass within the bounds.
336 est_error : float
337 3 times the standard error of the batch estimates.
338 n_samples : int
339 The number of samples actually used.
340 """
341 sn = max(1.0, np.sqrt(nu))
342 low = np.asarray(low, dtype=np.float64)
343 high = np.asarray(high, dtype=np.float64)
344 cho, lo, hi = _permuted_cholesky(covar, low / sn, high / sn)
345 n = cho.shape[0]
346 q, n_qmc_samples = _cbc_lattice(n, max(m // n_batches, 1))
347 rndm = rng.random(size=(n_batches, n))
348 prob, est_error, n_samples = _qmvt_inner(
349 q, rndm, int(n_qmc_samples), int(n_batches), cho, lo, hi, float(nu)
350 )
351 return prob, est_error, n_samples

354 def _permuted_cholesky(covar, low, high, tol=1e-10):
355 """Compute a scaled, permuted Cholesky factor, with integration bounds.
356  
357 The scaling and permuting of the dimensions accomplishes part of the
358 transformation of the original integration problem into a more numerically
359 tractable form. The lower-triangular Cholesky factor will then be used in
360 the subsequent integration. The integration bounds will be scaled and
361 permuted as well.
362  
363 Parameters
364 ----------
365 covar : (n, n) float array
366 Possibly singular, positive semidefinite symmetric covariance matrix.
367 low, high : (n,) float array
368 The low and high integration bounds.
369 tol : float, optional
370 The singularity tolerance.
371  
372 Returns
373 -------
374 cho : (n, n) float array
375 Lower Cholesky factor, scaled and permuted.
376 new_low, new_high : (n,) float array
377 The scaled and permuted low and high integration bounds.
378 """
379 # Make copies for outputting.
380 cho = np.array(covar, dtype=np.float64)
381 new_lo = np.array(low, dtype=np.float64)
382 new_hi = np.array(high, dtype=np.float64)
383 n = cho.shape[0]
384 if cho.shape != (n, n):
385 raise ValueError("expected a square symmetric array")
386 if new_lo.shape != (n,) or new_hi.shape != (n,):
387 raise ValueError(
388 "expected integration boundaries the same dimensions "
389 "as the covariance matrix"
390 )
391 # Scale by the sqrt of the diagonal.
392 dc = np.sqrt(np.maximum(np.diag(cho), 0.0))
393 # But don't divide by 0.
394 dc[dc == 0.0] = 1.0
395 new_lo /= dc
396 new_hi /= dc
397 cho /= dc
398 cho /= dc[:, np.newaxis]
399  
400 y = np.zeros(n)
401 sqtp = np.sqrt(2 * np.pi)
402 for k in range(n):
403 epk = (k + 1) * tol
404 im = k
405 ck = 0.0
406 dem = 1.0
407 s = 0.0
408 lo_m = 0.0
409 hi_m = 0.0
410 for i in range(k, n):
411 if cho[i, i] > tol:
412 ci = np.sqrt(cho[i, i])
413 if i > 0:
414 s = cho[i, :k] @ y[:k]
415 lo_i = (new_lo[i] - s) / ci
416 hi_i = (new_hi[i] - s) / ci
417 de = phi(hi_i) - phi(lo_i)
418 if de <= dem:
419 ck = ci
420 dem = de
421 lo_m = lo_i
422 hi_m = hi_i
423 im = i
424 if im > k:
425 # Swap im and k
426 cho[im, im] = cho[k, k]
427 _swap_slices(cho, np.s_[im, :k], np.s_[k, :k])
428 _swap_slices(cho, np.s_[im + 1:, im], np.s_[im + 1:, k])
429 _swap_slices(cho, np.s_[k + 1:im, k], np.s_[im, k + 1:im])
430 _swap_slices(new_lo, k, im)
431 _swap_slices(new_hi, k, im)
432 if ck > epk:
433 cho[k, k] = ck
434 cho[k, k + 1:] = 0.0
435 for i in range(k + 1, n):
436 cho[i, k] /= ck
437 cho[i, k + 1:i + 1] -= cho[i, k] * cho[k + 1:i + 1, k]
438 if abs(dem) > tol:
439 y[k] = ((np.exp(-lo_m * lo_m / 2) - np.exp(-hi_m * hi_m / 2)) /
440 (sqtp * dem))
441 else:
442 y[k] = (lo_m + hi_m) / 2
443 if lo_m < -10:
444 y[k] = hi_m
445 elif hi_m > 10:
446 y[k] = lo_m
447 cho[k, :k + 1] /= ck
448 new_lo[k] /= ck
449 new_hi[k] /= ck
450 else:
451 cho[k:, k] = 0.0
452 y[k] = (new_lo[k] + new_hi[k]) / 2
453 return cho, new_lo, new_hi

457 t = x[slc1].copy()
458 x[slc1] = x[slc2].copy()
459 x[slc2] = t
