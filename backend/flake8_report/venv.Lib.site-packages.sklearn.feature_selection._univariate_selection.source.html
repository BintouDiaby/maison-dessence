41 def f_oneway(*args):
42 """Perform a 1-way ANOVA.
43  
44 The one-way ANOVA tests the null hypothesis that 2 or more groups have
45 the same population mean. The test is applied to samples from two or
46 more groups, possibly with differing sizes.
47  
48 Read more in the :ref:`User Guide <univariate_feature_selection>`.
49  
50 Parameters
51 ----------
52 *args : {array-like, sparse matrix}
53 Sample1, sample2... The sample measurements should be given as
54 arguments.
55  
56 Returns
57 -------
58 f_statistic : float
59 The computed F-value of the test.
60 p_value : float
61 The associated p-value from the F-distribution.
62  
63 Notes
64 -----
65 The ANOVA test has important assumptions that must be satisfied in order
66 for the associated p-value to be valid.
67  
68 1. The samples are independent
69 2. Each sample is from a normally distributed population
70 3. The population standard deviations of the groups are all equal. This
71 property is known as homoscedasticity.
72  
73 If these assumptions are not true for a given set of data, it may still be
74 possible to use the Kruskal-Wallis H-test (`scipy.stats.kruskal`_) although
75 with some loss of power.
76  
77 The algorithm is from Heiman[2], pp.394-7.
78  
79 See ``scipy.stats.f_oneway`` that should give the same results while
80 being less efficient.
81  
82 References
83 ----------
84 .. [1] Lowry, Richard. "Concepts and Applications of Inferential
85 Statistics". Chapter 14.
86 http://vassarstats.net/textbook
87  
88 .. [2] Heiman, G.W. Research Methods in Statistics. 2002.
89 """
90 n_classes = len(args)
91 args = [as_float_array(a) for a in args]
92 n_samples_per_class = np.array([a.shape[0] for a in args])
93 n_samples = np.sum(n_samples_per_class)
94 ss_alldata = sum(safe_sqr(a).sum(axis=0) for a in args)
95 sums_args = [np.asarray(a.sum(axis=0)) for a in args]
96 square_of_sums_alldata = sum(sums_args) ** 2
97 square_of_sums_args = [s**2 for s in sums_args]
98 sstot = ss_alldata - square_of_sums_alldata / float(n_samples)
99 ssbn = 0.0
100 for k, _ in enumerate(args):
101 ssbn += square_of_sums_args[k] / n_samples_per_class[k]
102 ssbn -= square_of_sums_alldata / float(n_samples)
103 sswn = sstot - ssbn
104 dfbn = n_classes - 1
105 dfwn = n_samples - n_classes
106 msb = ssbn / float(dfbn)
107 msw = sswn / float(dfwn)
108 constant_features_idx = np.where(msw == 0.0)[0]
109 if np.nonzero(msb)[0].size != msb.size and constant_features_idx.size:

111 f = msb / msw
112 # flatten matrix to vector in sparse case
113 f = np.asarray(f).ravel()
114 prob = special.fdtrc(dfbn, dfwn, f)
115 return f, prob

174 def _chisquare(f_obs, f_exp):
175 """Fast replacement for scipy.stats.chisquare.
176  
177 Version from https://github.com/scipy/scipy/pull/2525 with additional
178 optimizations.
179 """
180 f_obs = np.asarray(f_obs, dtype=np.float64)
181  
182 k = len(f_obs)
183 # Reuse f_obs for chi-squared statistics
184 chisq = f_obs
185 chisq -= f_exp
186 chisq **= 2
187 with np.errstate(invalid="ignore"):
188 chisq /= f_exp
189 chisq = chisq.sum(axis=0)
190 return chisq, special.chdtrc(k - 1, chisq)
