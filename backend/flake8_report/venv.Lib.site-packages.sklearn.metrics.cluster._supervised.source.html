96 def contingency_matrix(
97 labels_true, labels_pred, *, eps=None, sparse=False, dtype=np.int64
98 ):
99 """Build a contingency matrix describing the relationship between labels.
100  
101 Read more in the :ref:`User Guide <contingency_matrix>`.
102  
103 Parameters
104 ----------
105 labels_true : array-like of shape (n_samples,)
106 Ground truth class labels to be used as a reference.
107  
108 labels_pred : array-like of shape (n_samples,)
109 Cluster labels to evaluate.
110  
111 eps : float, default=None
112 If a float, that value is added to all values in the contingency
113 matrix. This helps to stop NaN propagation.
114 If ``None``, nothing is adjusted.
115  
116 sparse : bool, default=False
117 If `True`, return a sparse CSR contingency matrix. If `eps` is not
118 `None` and `sparse` is `True` will raise ValueError.
119  
120 .. versionadded:: 0.18
121  
122 dtype : numeric type, default=np.int64
123 Output dtype. Ignored if `eps` is not `None`.
124  
125 .. versionadded:: 0.24
126  
127 Returns
128 -------
129 contingency : {array-like, sparse}, shape=[n_classes_true, n_classes_pred]
130 Matrix :math:`C` such that :math:`C_{i, j}` is the number of samples in
131 true class :math:`i` and in predicted class :math:`j`. If
132 ``eps is None``, the dtype of this array will be integer unless set
133 otherwise with the ``dtype`` argument. If ``eps`` is given, the dtype
134 will be float.
135 Will be a ``sklearn.sparse.csr_matrix`` if ``sparse=True``.
136  
137 Examples
138 --------
139 >>> from sklearn.metrics.cluster import contingency_matrix
140 >>> labels_true = [0, 0, 1, 1, 2, 2]
141 >>> labels_pred = [1, 0, 2, 1, 0, 2]
142 >>> contingency_matrix(labels_true, labels_pred)
143 array([[1, 1, 0],
144 [0, 1, 1],
145 [1, 0, 1]])
146 """
147  
148 if eps is not None and sparse:
149 raise ValueError("Cannot set 'eps' when sparse=True")
150  
151 classes, class_idx = np.unique(labels_true, return_inverse=True)
152 clusters, cluster_idx = np.unique(labels_pred, return_inverse=True)
153 n_classes = classes.shape[0]
154 n_clusters = clusters.shape[0]
155 # Using coo_matrix to accelerate simple histogram calculation,
156 # i.e. bins are consecutive integers
157 # Currently, coo_matrix is faster than histogram2d for simple cases
158 contingency = sp.coo_matrix(
159 (np.ones(class_idx.shape[0]), (class_idx, cluster_idx)),
160 shape=(n_classes, n_clusters),
161 dtype=dtype,
162 )
163 if sparse:
164 contingency = contingency.tocsr()
165 contingency.sum_duplicates()
166 else:
167 contingency = contingency.toarray()
168 if eps is not None:
169 # don't use += as contingency is integer
170 contingency = contingency + eps
171 return contingency

184 def pair_confusion_matrix(labels_true, labels_pred):
185 """Pair confusion matrix arising from two clusterings.
186  
187 The pair confusion matrix :math:`C` computes a 2 by 2 similarity matrix
188 between two clusterings by considering all pairs of samples and counting
189 pairs that are assigned into the same or into different clusters under
190 the true and predicted clusterings [1]_.
191  
192 Considering a pair of samples that is clustered together a positive pair,
193 then as in binary classification the count of true negatives is
194 :math:`C_{00}`, false negatives is :math:`C_{10}`, true positives is
195 :math:`C_{11}` and false positives is :math:`C_{01}`.
196  
197 Read more in the :ref:`User Guide <pair_confusion_matrix>`.
198  
199 Parameters
200 ----------
201 labels_true : array-like of shape (n_samples,), dtype=integral
202 Ground truth class labels to be used as a reference.
203  
204 labels_pred : array-like of shape (n_samples,), dtype=integral
205 Cluster labels to evaluate.
206  
207 Returns
208 -------
209 C : ndarray of shape (2, 2), dtype=np.int64
210 The contingency matrix.
211  
212 See Also
213 --------
214 sklearn.metrics.rand_score : Rand Score.
215 sklearn.metrics.adjusted_rand_score : Adjusted Rand Score.
216 sklearn.metrics.adjusted_mutual_info_score : Adjusted Mutual Information.
217  
218 References
219 ----------
220 .. [1] :doi:`Hubert, L., Arabie, P. "Comparing partitions."
221 Journal of Classification 2, 193–218 (1985).
222 <10.1007/BF01908075>`
223  
224 Examples
225 --------
226 Perfectly matching labelings have all non-zero entries on the
227 diagonal regardless of actual label values:
228  
229 >>> from sklearn.metrics.cluster import pair_confusion_matrix
230 >>> pair_confusion_matrix([0, 0, 1, 1], [1, 1, 0, 0])
231 array([[8, 0],
232 [0, 4]]...
233  
234 Labelings that assign all classes members to the same clusters
235 are complete but may be not always pure, hence penalized, and
236 have some off-diagonal non-zero entries:
237  
238 >>> pair_confusion_matrix([0, 0, 1, 2], [0, 0, 1, 1])
239 array([[8, 2],
240 [0, 2]]...
241  
242 Note that the matrix is not symmetric.
243 """
244 labels_true, labels_pred = check_clusterings(labels_true, labels_pred)
245 n_samples = np.int64(labels_true.shape[0])
246  
247 # Computation using the contingency data
248 contingency = contingency_matrix(
249 labels_true, labels_pred, sparse=True, dtype=np.int64
250 )
251 n_c = np.ravel(contingency.sum(axis=1))
252 n_k = np.ravel(contingency.sum(axis=0))
253 sum_squares = (contingency.data**2).sum()
254 C = np.empty((2, 2), dtype=np.int64)
255 C[1, 1] = sum_squares - n_samples
256 C[0, 1] = contingency.dot(n_k).sum() - sum_squares
257 C[1, 0] = contingency.transpose().dot(n_c).sum() - sum_squares
258 C[0, 0] = n_samples**2 - C[0, 1] - C[1, 0] - sum_squares
259 return C

270 """Rand index.
271  
272 The Rand Index computes a similarity measure between two clusterings
273 by considering all pairs of samples and counting pairs that are
274 assigned in the same or different clusters in the predicted and
275 true clusterings [1]_ [2]_.
276  
277 The raw RI score [3]_ is:
278  
279 .. code-block:: text
280  
281 RI = (number of agreeing pairs) / (number of pairs)
282  
283 Read more in the :ref:`User Guide <rand_score>`.
284  
285 Parameters
286 ----------
287 labels_true : array-like of shape (n_samples,), dtype=integral
288 Ground truth class labels to be used as a reference.
289  
290 labels_pred : array-like of shape (n_samples,), dtype=integral
291 Cluster labels to evaluate.
292  
293 Returns
294 -------
295 RI : float
296 Similarity score between 0.0 and 1.0, inclusive, 1.0 stands for
297 perfect match.
298  
299 See Also

301 adjusted_rand_score: Adjusted Rand Score.
302 adjusted_mutual_info_score: Adjusted Mutual Information.
303  
304 References

306 .. [1] :doi:`Hubert, L., Arabie, P. "Comparing partitions."
307 Journal of Classification 2, 193–218 (1985).
308 <10.1007/BF01908075>`.
309  
310 .. [2] `Wikipedia: Simple Matching Coefficient
311 <https://en.wikipedia.org/wiki/Simple_matching_coefficient>`_
312  
313 .. [3] `Wikipedia: Rand Index <https://en.wikipedia.org/wiki/Rand_index>`_
314  
315 Examples

317 Perfectly matching labelings have a score of 1 even
318  
319 >>> from sklearn.metrics.cluster import rand_score

327 0.83

351  
352 The Rand Index computes a similarity measure between two clusterings

356  
357 The raw RI score is then "adjusted for chance" into the ARI score
358 using the following scheme::
359  
360 ARI = (RI - Expected_RI) / (max(RI) - Expected_RI)
