308 def sample(self, n_samples=1, random_state=None):
309 """Generate random samples from the model.
310  
311 Currently, this is implemented only for gaussian and tophat kernels.
312  
313 Parameters
314 ----------
315 n_samples : int, default=1
316 Number of samples to generate.
317  
318 random_state : int, RandomState instance or None, default=None
319 Determines random number generation used to generate
320 random samples. Pass an int for reproducible results
321 across multiple function calls.
322 See :term:`Glossary <random_state>`.
323  
324 Returns
325 -------
326 X : array-like of shape (n_samples, n_features)
327 List of samples.
328 """
329 check_is_fitted(self)
330 # TODO: implement sampling for other valid kernel shapes
331 if self.kernel not in ["gaussian", "tophat"]:
332 raise NotImplementedError()
333  
334 data = np.asarray(self.tree_.data)
335  
336 rng = check_random_state(random_state)
337 u = rng.uniform(0, 1, size=n_samples)
338 if self.tree_.sample_weight is None:
339 i = (u * data.shape[0]).astype(np.int64)
340 else:
341 cumsum_weight = np.cumsum(np.asarray(self.tree_.sample_weight))
342 sum_weight = cumsum_weight[-1]
343 i = np.searchsorted(cumsum_weight, u * sum_weight)
344 if self.kernel == "gaussian":
345 return np.atleast_2d(rng.normal(data[i], self.bandwidth_))
346  
347 elif self.kernel == "tophat":
348 # we first draw points from a d-dimensional normal distribution,
349 # then use an incomplete gamma function to map them to a uniform
350 # d-dimensional tophat distribution.
351 dim = data.shape[1]
352 X = rng.normal(size=(n_samples, dim))
353 s_sq = row_norms(X, squared=True)
354 correction = (
355 gammainc(0.5 * dim, 0.5 * s_sq) ** (1.0 / dim)
356 * self.bandwidth_
357 / np.sqrt(s_sq)
358 )
359 return data[i] + X * correction[:, np.newaxis]
