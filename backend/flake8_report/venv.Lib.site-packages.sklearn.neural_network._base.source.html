67 def inplace_softmax(X):
68 """Compute the K-way softmax function inplace.
69  
70 Parameters
71 ----------
72 X : {array-like, sparse matrix}, shape (n_samples, n_features)
73 The input data.
74 """
75 tmp = X - X.max(axis=1)[:, np.newaxis]
76 np.exp(tmp, out=X)
77 X /= X.sum(axis=1)[:, np.newaxis]

105 def inplace_logistic_derivative(Z, delta):
106 """Apply the derivative of the logistic sigmoid function.
107  
108 It exploits the fact that the derivative is a simple function of the output
109 value from logistic function.
110  
111 Parameters
112 ----------
113 Z : {array-like, sparse matrix}, shape (n_samples, n_features)
114 The data which was output from the logistic activation function during
115 the forward pass.
116  
117 delta : {array-like}, shape (n_samples, n_features)
118 The backpropagated error signal to be modified inplace.
119 """
120 delta *= Z
121 delta *= 1 - Z

124 def inplace_tanh_derivative(Z, delta):
125 """Apply the derivative of the hyperbolic tanh function.

129  
130 Parameters

132 Z : {array-like, sparse matrix}, shape (n_samples, n_features)
