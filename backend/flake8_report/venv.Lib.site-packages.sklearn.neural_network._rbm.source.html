25 class BernoulliRBM(ClassNamePrefixFeaturesOutMixin, TransformerMixin, BaseEstimator):
26 """Bernoulli Restricted Boltzmann Machine (RBM).
27  
28 A Restricted Boltzmann Machine with binary visible units and
29 binary hidden units. Parameters are estimated using Stochastic Maximum
30 Likelihood (SML), also known as Persistent Contrastive Divergence (PCD)
31 [2].
32  
33 The time complexity of this implementation is ``O(d ** 2)`` assuming
34 d ~ n_features ~ n_components.
35  
36 Read more in the :ref:`User Guide <rbm>`.
37  
38 Parameters
39 ----------
40 n_components : int, default=256
41 Number of binary hidden units.
42  
43 learning_rate : float, default=0.1
44 The learning rate for weight updates. It is *highly* recommended
45 to tune this hyper-parameter. Reasonable values are in the
46 10**[0., -3.] range.
47  
48 batch_size : int, default=10
49 Number of examples per minibatch.
50  
51 n_iter : int, default=10
52 Number of iterations/sweeps over the training dataset to perform
53 during training.
54  
55 verbose : int, default=0
56 The verbosity level. The default, zero, means silent mode. Range
57 of values is [0, inf].
58  
59 random_state : int, RandomState instance or None, default=None
60 Determines random number generation for:
61  
62 - Gibbs sampling from visible and hidden layers.
63  
64 - Initializing components, sampling from layers during fit.
65  
66 - Corrupting the data when scoring samples.
67  
68 Pass an int for reproducible results across multiple function calls.
69 See :term:`Glossary <random_state>`.
70  
71 Attributes
72 ----------
73 intercept_hidden_ : array-like of shape (n_components,)
74 Biases of the hidden units.
75  
76 intercept_visible_ : array-like of shape (n_features,)
77 Biases of the visible units.
78  
79 components_ : array-like of shape (n_components, n_features)
80 Weight matrix, where `n_features` is the number of
81 visible units and `n_components` is the number of hidden units.
82  
83 h_samples_ : array-like of shape (batch_size, n_components)
84 Hidden Activation sampled from the model distribution,
85 where `batch_size` is the number of examples per minibatch and
86 `n_components` is the number of hidden units.
87  
88 n_features_in_ : int
89 Number of features seen during :term:`fit`.
90  
91 .. versionadded:: 0.24
92  
93 feature_names_in_ : ndarray of shape (`n_features_in_`,)
94 Names of features seen during :term:`fit`. Defined only when `X`
95 has feature names that are all strings.
96  
97 .. versionadded:: 1.0

234 def _free_energy(self, v):
235 """Computes the free energy F(v) = - log sum_h exp(-E(v,h)).
236  
237 Parameters
238 ----------
239 v : ndarray of shape (n_samples, n_features)
240 Values of the visible layer.
241  
242 Returns
243 -------
244 free_energy : ndarray of shape (n_samples,)
245 The value of the free energy.
246 """
247 return -safe_sparse_dot(v, self.intercept_visible_) - np.logaddexp(

314 def _fit(self, v_pos, rng):
315 """Inner fit for one mini-batch.
316  
317 Adjust the parameters to maximize the likelihood of v using
318 Stochastic Maximum Likelihood (SML).
319  
320 Parameters
321 ----------
322 v_pos : ndarray of shape (n_samples, n_features)
323 The data to use for training.
324  
325 rng : RandomState instance
326 Random number generator to use for sampling.
327 """
328 h_pos = self._mean_hiddens(v_pos)
329 v_neg = self._sample_visibles(self.h_samples_, rng)
330 h_neg = self._mean_hiddens(v_neg)
331  
332 lr = float(self.learning_rate) / v_pos.shape[0]
333 update = safe_sparse_dot(v_pos.T, h_pos, dense_output=True).T
334 update -= np.dot(h_neg.T, v_neg)
335 self.components_ += lr * update
336 self.intercept_hidden_ += lr * (h_pos.sum(axis=0) - h_neg.sum(axis=0))
337 self.intercept_visible_ += lr * (
338 np.asarray(v_pos.sum(axis=0)).squeeze() - v_neg.sum(axis=0)
339 )
340  
341 h_neg[rng.uniform(size=h_neg.shape) < h_neg] = 1.0 # sample binomial

344 def score_samples(self, X):
345 """Compute the pseudo-likelihood of X.
346  
347 Parameters
348 ----------
349 X : {array-like, sparse matrix} of shape (n_samples, n_features)
350 Values of the visible layer. Must be all-boolean (not checked).
351  
352 Returns
353 -------
354 pseudo_likelihood : ndarray of shape (n_samples,)
355 Value of the pseudo-likelihood (proxy for likelihood).
356  
357 Notes
358 -----
359 This method is not deterministic: it computes a quantity called the
360 free energy on X, then on a randomly corrupted version of X, and
361 returns the log of the logistic function of the difference.
362 """
363 check_is_fitted(self)
364  
365 v = validate_data(self, X, accept_sparse="csr", reset=False)
366 rng = check_random_state(self.random_state)
367  
368 # Randomly corrupt one feature in each sample in v.
369 ind = (np.arange(v.shape[0]), rng.randint(0, v.shape[1], v.shape[0]))
370 if sp.issparse(v):
371 data = -2 * v[ind] + 1
372 if isinstance(data, np.matrix): # v is a sparse matrix
373 v_ = v + sp.csr_matrix((data.A.ravel(), ind), shape=v.shape)
374 else: # v is a sparse array
375 v_ = v + sp.csr_array((data.ravel(), ind), shape=v.shape)
376 else:
377 v_ = v.copy()
378 v_[ind] = 1 - v_[ind]

380 fe = self._free_energy(v)
381 fe_ = self._free_energy(v_)
382 # log(expit(x)) = log(1 / (1 + exp(-x)) = -np.logaddexp(0, -x)
