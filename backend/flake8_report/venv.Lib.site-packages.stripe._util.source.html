174 def secure_compare(val1, val2):
175 """
176 Returns True if the two strings are equal, False otherwise.
177 The time taken is independent of the number of characters that match.
178 For the sake of simplicity, this function executes in constant time
179 only when the two strings have the same length. It short-circuits when
180 they have different lengths.
181 """
182 if len(val1) != len(val2):
183 return False
184 result = 0
185 if isinstance(val1, bytes) and isinstance(val2, bytes):
186 for x, y in zip(val1, val2):
187 result |= x ^ y
188 else:
189 for x, y in zip(val1, val2):
190 result |= ord(cast(str, x)) ^ ord(cast(str, y))
191 return result == 0
